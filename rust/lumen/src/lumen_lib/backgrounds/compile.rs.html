48/// One compiled shader pass.
49#[derive(Debug, Clone)]
50pub struct CompiledPass {
51    pub id: &'static str,
52    pub feedback: bool,
53    /// Name of the combined sampler uniform in the emitted GLSL, if any.
54    pub sampler: Option<String>,
55    /// Name of the uniform block emitted by naga (e.g. `Params_block_0Fragment`).
56    pub uniform_block: String,
57    pub glsl: String,
58}

67fn compile_wgsl(source: &str, stage: ShaderStage) -> Result<String, String> {
68    let module =
69        naga::front::wgsl::parse_str(source).map_err(|error| format!("WGSL parse error:\n{error}"))?;
70
71    let mut validator = naga::valid::Validator::new(
72        naga::valid::ValidationFlags::all(),
73        naga::valid::Capabilities::empty(),
74    );
75    let info = validator
76        .validate(&module)
77        .map_err(|error| format!("WGSL validation error:\n{error}"))?;
78
79    let options = naga::back::glsl::Options {
80        version: naga::back::glsl::Version::Embedded {
81            version: 300,
82            is_webgl: true,
83        },
84        ..Default::default()
85    };
86    let pipeline = naga::back::glsl::PipelineOptions {
87        shader_stage: stage,
88        entry_point: "main".to_string(),
89        multiview: None,
90    };
91
92    let mut output = String::new();
93    let mut writer = naga::back::glsl::Writer::new(
94        &mut output,
95        &module,
96        &info,
97        &options,
98        &pipeline,
99        Default::default(),
100    )
101    .map_err(|error| format!("GLSL writer error:\n{error}"))?;
102    writer
103        .write()
104        .map_err(|error| format!("GLSL emit error:\n{error}"))?;
105    Ok(output)
106}

108/// Finds the name of the uniform block emitted by naga.
109fn uniform_block(glsl: &str) -> Option<String> {
110    for (index, _) in glsl.match_indices("uniform ") {
111        let rest = &glsl[index + "uniform ".len()..];
112        let name: String = rest
113            .chars()
114            .take_while(|c| c.is_ascii_alphanumeric() || *c == '_')
115            .collect();
116        if name.is_empty() {
117            continue;
118        }
119        if rest[name.len()..].trim_start().starts_with('{') {
120            return Some(name);
121        }
122    }
123    None
124}

126/// Finds the name of the combined sampler uniform emitted by naga.
127fn sampler_uniform(glsl: &str) -> Option<String> {
128    glsl.lines().find_map(|line| {
129        let rest = line.trim().strip_prefix("uniform ")?;
130        let rest = rest
131            .strip_prefix("highp ")
132            .or_else(|| rest.strip_prefix("mediump "))
133            .or_else(|| rest.strip_prefix("lowp "))
134            .unwrap_or(rest);
135        let rest = rest.strip_prefix("sampler2D ")?;
136        let name = rest.trim_end_matches(';').trim();
137        (!name.is_empty()).then(|| name.to_string())
138    })
139}

141/// Compiles every catalog effect. Fails on the first shader error.
142pub fn compile_catalog() -> Result<Vec<CompiledEffect>, String> {
143    catalog::catalog()
144        .into_iter()
145        .map(|def| {
146            let mut passes = Vec::new();
147            for pass in &def.passes {
148                let source = format!("{PRELUDE}\n{}", pass.source);
149                let glsl = compile_wgsl(&source, ShaderStage::Fragment).map_err(|error| {
150                    format!("effect `{}` pass `{}`:\n{error}", def.id, pass.id)
151                })?;
152
153                let block = uniform_block(&glsl).ok_or_else(|| {
154                    format!(
155                        "effect `{}` pass `{}` emitted no uniform block",
156                        def.id, pass.id
157                    )
158                })?;
159
160                passes.push(CompiledPass {
161                    id: pass.id,
162                    feedback: pass.feedback,
163                    sampler: sampler_uniform(&glsl),
164                    uniform_block: block,
165                    glsl,
166                });
167            }
168            Ok(CompiledEffect { def, passes })
169        })
170        .collect()
171}

173/// Builds the TypeScript module consumed by the webview.
174pub fn generate_typescript() -> Result<String, String> {
175    let vertex_glsl = compile_wgsl(VERTEX, ShaderStage::Vertex)?;
176    let vertex_block = uniform_block(&vertex_glsl);
177
178    let display_glsl = compile_wgsl(&format!("{PRELUDE}\n{DISPLAY}"), ShaderStage::Fragment)?;
179    let display_sampler = sampler_uniform(&display_glsl);
180    let display_block = uniform_block(&display_glsl);
181
182    let compiled = compile_catalog()?;
183    let effects: Vec<Value> = compiled
184        .iter()
185        .map(|compiled_effect| {
186            let def = &compiled_effect.def;
187            let passes: Vec<Value> = compiled_effect
188                .passes
189                .iter()
190                .map(|pass| {
191                    json!({
192                        "id": pass.id,
193                        "feedback": pass.feedback,
194                        "glsl": &pass.glsl,
195                        "samplerUniform": &pass.sampler,
196                        "uniformBlock": &pass.uniform_block,
197                    })
198                })
199                .collect();
200
201            json!({
202                "id": def.id,
203                "name": def.name,
204                "category": def.category,
205                "tier": def.tier,
206                "version": def.version,
207                "cost": def.cost,
208                "params": &def.params,
209                "passes": passes,
210            })
211        })
212        .collect();
213
214    let layout: UniformLayout = catalog::layout();
215    let bundle = json!({
216        "vertex": {
217            "glsl": vertex_glsl,
218            "uniformBlock": vertex_block,
219        },
220        "display": {
221            "glsl": display_glsl,
222            "samplerUniform": display_sampler,
223            "uniformBlock": display_block,
224        },
225        "layout": layout,
226        "effects": effects,
227    });
228
229    let body = serde_json::to_string_pretty(&bundle).map_err(|error| error.to_string())?;
230    Ok(format!(
231        "// AUTO-GENERATED by `cargo run --example bg_compile`.\n\
232         // Source of truth: src-tauri/src/backgrounds. Do not edit by hand.\n\
233         \n\
234         export const BACKGROUND_BUNDLE = {body} as const;\n\
235         \n\
236         export type BackgroundBundle = typeof BACKGROUND_BUNDLE;\n"
237    ))
238}

240/// Writes the generated bundle into `src/backgrounds/generated/catalog.ts`.
241pub fn write_generated() -> Result<String, String> {
242    let output = generate_typescript()?;
243    let path = Path::new(env!("CARGO_MANIFEST_DIR"))
244        .join("..")
245        .join("src")
246        .join("backgrounds")
247        .join("generated")
248        .join("catalog.ts");
249
250    if let Some(parent) = path.parent() {
251        std::fs::create_dir_all(parent).map_err(|error| error.to_string())?;
252    }
253    std::fs::write(&path, output).map_err(|error| error.to_string())?;
254    Ok(path.display().to_string())
255}
