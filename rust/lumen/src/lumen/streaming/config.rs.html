1//! Persistent streaming settings and the runtime status snapshot.
2//!
3//! [`StreamingConfig`] is the user-editable configuration stored as
4//! `config/streaming.json` under the app base directory. It is round-tripped
5//! through the Tauri commands in [`crate::streaming::manager`] and is always
6//! passed through [`sanitize_config`] before being saved or applied, so out of
7//! range values coming from the frontend can never reach the runtime.
8//!
9//! [`StreamingStatus`] is a read-only projection of the live
10//! [`crate::streaming::manager::StreamManager`] (subscriber counts, mobile
11//! connection state, HTML server URL and current volumes) that the frontend
12//! polls or receives through the `streaming_status_changed` event.
13//!
14//! All fallible functions return `Result<_, String>` carrying a raw I/O or
15//! serialization error message.
16

21/// User-editable streaming configuration persisted to `config/streaming.json`.
22///
23/// The struct is serialized to and from the frontend, so field names use
24/// `snake_case`. Values are normalized by [`sanitize_config`] on load and on
25/// every update.
26#[derive(Debug, Clone, Serialize, Deserialize)]
27pub struct StreamingConfig {
28    /// Whether remote subscribers may open the desktop `preview` stream.
29    pub preview_enabled: bool,
30    /// Target frame rate for the main output. Valid values are `1`, `15`,
31    /// `24`, `30` and `60`; any other value falls back to `1`.
32    pub main_fps: u8,
33    /// Target resolution for the main output. Valid values are `"720p"`,
34    /// `"1080p"`, `"1440p"` and `"4K"`; any other value falls back to
35    /// `"1080p"`.
36    pub main_resolution: String,
37    /// Whether the local HTML presentation server should be running.
38    pub html_server_enabled: bool,
39    /// TCP port bound by the HTML presentation server on all interfaces.
40    /// `0` is normalized to the default `8090`.
41    pub html_server_port: u16,
42    /// Whether hardware video encoding should be preferred when available.
43    pub hardware_encoding: bool,
44    /// Master switch for content protection (e.g. keeping protected content
45    /// out of captures). Effective only while also enabled at runtime through
46    /// [`crate::streaming::manager::set_stream_content_protected`].
47    pub content_protection: bool,
48    /// Master output volume, `0..=100`. Defaults to `80` when absent from the
49    /// stored JSON.
50    #[serde(default = "default_master_volume")]
51    pub master_volume: u8,
52    /// Per-device output volumes keyed by mobile device id, each in
53    /// `0..=100`. Missing ids are treated as using the master volume.
54    #[serde(default)]
55    pub device_volumes: HashMap<String, u8>,
56}
57
58/// Default master volume applied when the persisted config omits the field.
59fn default_master_volume() -> u8 {
60    80
61}
62
63impl Default for StreamingConfig {
64    fn default() -> Self {
65        Self {
66            preview_enabled: true,
67            main_fps: 1,
68            main_resolution: "1080p".to_string(),
69            html_server_enabled: false,
70            html_server_port: 8090,
71            hardware_encoding: false,
72            content_protection: true,
73            master_volume: 80,
74            device_volumes: HashMap::new(),
75        }
76    }
77}

79/// Read-only snapshot of the live streaming state sent to the frontend.
80///
81/// Produced by [`crate::streaming::manager::StreamManager::status`] and
82/// forwarded whenever the state changes, either as the return value of the
83/// `get_streaming_status` command or through the `streaming_status_changed`
84/// Tauri event.
85#[derive(Debug, Clone, Serialize, Deserialize)]
86pub struct StreamingStatus {
87    /// Number of sessions subscribed to the desktop `preview` or
88    /// `app_preview` streams, saturated at [`u8::MAX`].
89    pub preview_subs: u8,
90    /// Number of sessions subscribed to the `main` output, saturated at
91    /// [`u8::MAX`].
92    pub main_subs: u8,
93    /// Whether at least one mobile device is currently sending media.
94    pub mobile_connected: bool,
95    /// Whether the HTML presentation server is currently running.
96    pub html_active: bool,
97    /// URL of the HTML presentation page, or `None` while the server is
98    /// stopped or the local IP address cannot be resolved.
99    pub html_url: Option<String>,
100    /// Current master output volume, `0..=100`.
101    #[serde(default = "default_master_volume")]
102    pub master_volume: u8,
103    /// Current per-device output volumes keyed by mobile device id.
104    #[serde(default)]
105    pub device_volumes: HashMap<String, u8>,
106}
107
108impl Default for StreamingStatus {
109    fn default() -> Self {
110        Self {
111            preview_subs: 0,
112            main_subs: 0,
113            mobile_connected: false,
114            html_active: false,
115            html_url: None,
116            master_volume: 80,
117            device_volumes: HashMap::new(),
118        }
119    }
120}

122/// Clamps and repairs a [`StreamingConfig`] so it is safe to apply at runtime.
123///
124/// The normalization rules are:
125///
126/// * `main_fps` must be one of `1`, `15`, `24`, `30`, `60`, else it becomes
127///   `1`.
128/// * `main_resolution` must be one of `"720p"`, `"1080p"`, `"1440p"`,
129///   `"4K"`, else it becomes `"1080p"`.
130/// * `html_server_port` `0` becomes `8090`.
131/// * `master_volume` and every entry in `device_volumes` are clamped to
132///   `100`.
133///
134/// This function never fails; it always returns a usable configuration.
135pub fn sanitize_config(mut config: StreamingConfig) -> StreamingConfig {
136    if !matches!(config.main_fps, 1 | 15 | 24 | 30 | 60) {
137        config.main_fps = 1;
138    }
139
140    if !matches!(
141        config.main_resolution.as_str(),
142        "720p" | "1080p" | "1440p" | "4K"
143    ) {
144        config.main_resolution = "1080p".to_string();
145    }
146
147    if config.html_server_port == 0 {
148        config.html_server_port = 8090;
149    }
150
151    config.master_volume = config.master_volume.min(100);
152    for volume in config.device_volumes.values_mut() {
153        *volume = (*volume).min(100);
154    }
155
156    config
157}

159/// Creates the directory that stores the streaming config file.
160///
161/// Called during startup (see
162/// [`crate::streaming::manager::initialize_streaming_state`]) before the
163/// config is first written.
164///
165/// # Errors
166///
167/// Returns the underlying error message when the base directory cannot be
168/// resolved or the `config` directory cannot be created.
169pub fn ensure_streaming_storage() -> Result<(), String> {
170    let path = streaming_config_path()?;
171    if let Some(parent) = path.parent() {
172        fs::create_dir_all(parent).map_err(|error| error.to_string())?;
173    }
174    Ok(())
175}

177/// Loads the persisted [`StreamingConfig`], creating it if missing.
178///
179/// When `config/streaming.json` does not exist, the default configuration is
180/// written to disk and returned. When it exists, it is parsed and passed
181/// through [`sanitize_config`] before being returned.
182///
183/// # Errors
184///
185/// Returns the underlying error message when the path cannot be resolved, the
186/// file cannot be read or written, or the JSON is malformed.
187pub fn load_streaming_config() -> Result<StreamingConfig, String> {
188    let path = streaming_config_path()?;
189    if !path.exists() {
190        let default_config = StreamingConfig::default();
191        save_streaming_config(&default_config)?;
192        return Ok(default_config);
193    }
194
195    let content = fs::read_to_string(path).map_err(|error| error.to_string())?;
196    let parsed: StreamingConfig =
197        serde_json::from_str(&content).map_err(|error| error.to_string())?;
198    Ok(sanitize_config(parsed))
199}

201/// Serializes `config` as pretty JSON and writes it to
202/// `config/streaming.json`, replacing any previous contents.
203///
204/// The config file is stored in a `config` directory that may be written to at
205/// any time, so callers should ensure the returned value is already
206/// sanitized.
207///
208/// # Errors
209///
210/// Returns the underlying error message when the path cannot be resolved,
211/// serialization fails, or the file cannot be written.
212pub fn save_streaming_config(config: &StreamingConfig) -> Result<(), String> {
213    let path = streaming_config_path()?;
214    let content = serde_json::to_string_pretty(config).map_err(|error| error.to_string())?;
215    fs::write(path, content).map_err(|error| error.to_string())
216}

218/// Resolves the application base directory that hosts all Lumen data.

223/// Returns the absolute path of the streaming config file
224/// (`<app_base_dir>/config/streaming.json`).
225fn streaming_config_path() -> Result<PathBuf, String> {
226    Ok(app_base_dir()?.join("config").join("streaming.json"))
