1//! Shared streaming state and the Tauri commands exposed to the frontend.
2//!
3//! This module owns [`StreamManager`], the single source of truth for every
4//! WebRTC artifact produced by the desktop app: local media tracks, the peer
5//! connections currently subscribed to each stream, connected mobile peers,
6//! per-device volumes and the HTML presentation server runtime. It is stored
7//! inside a [`StreamingState`] and registered as Tauri managed state during
8//! setup by [`initialize_streaming_state`].
9//!
10//! The functions annotated with `#[tauri::command]` at the bottom of the file
11//! are the **frontend API**: they are invoked from the desktop webview through
12//! `@tauri-apps/api` `invoke()` and mirror the TypeScript wrappers in
13//! `src/services/streaming-service.ts`. Remote devices never call these
14//! commands directly; they use the WebSocket signaling helpers in
15//! [`crate::streaming::signaling`].
16//!
17//! # Events emitted
18//!
19//! * `streaming_status_changed` - emitted by [`StreamManager::emit_status`]
20//!   after any mutation, carrying a [`StreamingStatus`] snapshot.
21//! * `mobile_stream_ended` - emitted by [`cleanup_session`] when a mobile
22//!   device stops sending, carrying `{ device_id }`.
23

49/// Internal bookkeeping for one connected mobile device.
50///
51/// Created by [`crate::streaming::signaling::handle_mobile_offer`] when the
52/// device sends its SDP offer and updated as tracks arrive.
53struct MobilePeer {
54    /// WebRTC connection used to receive (and relay) the device media.
55    pub peer: Arc<RTCPeerConnection>,
56    /// Stable device id supplied by the mobile client (see `crate::devices`).
57    pub device_id: String,
58    /// Set to `true` once at least one video track has been received.
59    pub has_video: bool,
60    /// Set to `true` once at least one audio track has been received.
61    pub has_audio: bool,
62    /// Last reported video orientation (`"portrait"` / `"landscape"`).
63    pub video_orientation: Option<String>,
64}

66/// Central runtime state for the live streaming subsystem.
67///
68/// One instance lives inside [`StreamingState`] for the whole lifetime of the
69/// app. All tracks and peer maps are protected by the [`tokio::sync::Mutex`]
70/// wrapping this struct, so methods that only read a single field still take
71/// `&self` while mutations take `&mut self`.
72pub struct StreamManager {
73    /// Handle used to emit Tauri events and resolve managed state.
74    pub app: AppHandle,
75    /// WebRTC API (media engine + interceptors) used to create peer
76    /// connections.
77    pub api: Arc<webrtc::api::API>,
78    /// H264 track for the desktop `preview` stream.
79    pub preview_track: Arc<TrackLocalStaticSample>,
80    /// H264 track for the in-app `app_preview` stream, fed by
81    /// [`crate::streaming::app_preview_producer`].
82    pub app_preview_track: Arc<TrackLocalStaticSample>,
83    /// H264 track for the `main` program output.
84    pub main_track: Arc<TrackLocalStaticSample>,
85    /// RTP passthrough track that relays H264 video received from a mobile
86    /// device to `mobile_preview` subscribers.
87    pub mobile_preview_video_h264_track: Arc<TrackLocalStaticRTP>,
88    /// Opus RTP tracks keyed by mobile device id, created on demand when a
89    /// device audio stream is requested.
90    pub device_audio_tracks: HashMap<String, Arc<TrackLocalStaticRTP>>,
91    /// Subscribers of the `preview` stream keyed by session id.
92    pub preview_peers: HashMap<String, Arc<RTCPeerConnection>>,
93    /// Subscribers of the `app_preview` stream keyed by session id.
94    pub app_preview_peers: HashMap<String, Arc<RTCPeerConnection>>,
95    /// Subscribers of the `main` stream keyed by session id.
96    pub main_peers: HashMap<String, Arc<RTCPeerConnection>>,
97    /// Subscribers of the `mobile_preview` relay keyed by session id.
98    pub mobile_preview_peers: HashMap<String, Arc<RTCPeerConnection>>,
99    /// Active mobile ingestion peers keyed by session id.
100    mobile_peers: HashMap<String, MobilePeer>,
101    /// Audio-only subscribers of a given mobile device, keyed by device id
102    /// and then by session id (`device_id -> session_id -> peer`).
103    pub mobile_audio_peers: HashMap<String, HashMap<String, Arc<RTCPeerConnection>>>,
104    /// When `Some(device_id)`, only that device's video is relayed to
105    /// `mobile_preview`; `None` forwards every device. Set by
106    /// [`set_mobile_preview_device`].
107    pub selected_mobile_preview_device: Option<String>,
108    /// Active configuration, kept in sync with the on-disk
109    /// [`StreamingConfig`].
110    pub config: StreamingConfig,
111    /// Whether protected content is currently hidden from streams. Requires
112    /// [`StreamingConfig::content_protection`] to be enabled.
113    pub is_content_protected: bool,
114    /// Runtime of the local HTML presentation server.
115    pub html_server: HtmlServerRuntime,
116}

118/// Tauri managed state wrapping the shared [`StreamManager`].
119///
120/// Obtained inside commands with `State<'_, StreamingState>`. The inner mutex
121/// must be locked with `.lock().await`; helpers such as
122/// [`crate::streaming::signaling::subscribe_stream`] clone the
123/// [`Arc`] to keep the lock scope short.
124#[derive(Clone)]
125pub struct StreamingState {
126    /// Shared manager guarded by an async mutex.
127    pub manager: Arc<Mutex<StreamManager>>,
128}

130impl StreamManager {
131    /// Builds a [`StreamingStatus`] snapshot from the current state.
132    ///
133    /// Subscriber counts are saturated at [`u8::MAX`] and the HTML server URL
134    /// is included only while the server is running and the local IP can be
135    /// resolved.
136    pub fn status(&self) -> StreamingStatus {
137        let total_preview_subs = self.preview_peers.len() + self.app_preview_peers.len();
138        StreamingStatus {
139            preview_subs: total_preview_subs.min(u8::MAX as usize) as u8,
140            main_subs: self.main_peers.len().min(u8::MAX as usize) as u8,
141            mobile_connected: !self.mobile_peers.is_empty(),
142            html_active: self.html_server.is_active(),
143            html_url: self.html_server.url(),
144            master_volume: self.config.master_volume,
145            device_volumes: self.config.device_volumes.clone(),
146        }
147    }
148
149    /// Emits the `streaming_status_changed` Tauri event with a fresh
150    /// [`Self::status`] snapshot.
151    ///
152    /// Called after every mutation that the frontend should observe.
153    /// Emission failures are ignored.
154    pub fn emit_status(&self) {
155        let _ = self.app.emit("streaming_status_changed", self.status());
156    }
157
158    /// Starts or stops the HTML presentation server to match
159    /// [`StreamingConfig::html_server_enabled`], binding
160    /// [`StreamingConfig::html_server_port`] when enabled.
161    ///
162    /// Safe to call repeatedly: [`HtmlServerRuntime::start`] is a no-op when
163    /// the server is already running on the requested port.
164    pub fn apply_html_server_config(&mut self) {
165        if self.config.html_server_enabled {
166            self.html_server.start(self.config.html_server_port);
167        } else {
168            self.html_server.stop();
169        }
170    }
171
172    /// Returns the WebRTC peer for a connected mobile device, or `None` if the
173    /// given session id is not found.
174    ///
175    /// Used by [`crate::streaming::signaling`] helpers (e.g.
176    /// [`crate::streaming::signaling::add_webrtc_ice_candidate`]) to reach
177    /// mobile ingestion peers.
178    pub fn get_mobile_peer(&self, session_id: &str) -> Option<Arc<RTCPeerConnection>> {
179        self.mobile_peers
180            .get(session_id)
181            .map(|entry| entry.peer.clone())
182    }
183
184    /// Updates track reception flags for a mobile peer.
185    ///
186    /// Only flips `has_video` or `has_audio` to `true` if the corresponding
187    /// argument is `true`. Returns the new `(has_video, has_audio,
188    /// video_orientation)` triple for the peer, or `None` if the session was
189    /// not found.
190    ///
191    /// Called by the mobile track handler in [`crate::streaming::signaling`].
192    pub fn set_mobile_track_state(
193        &mut self,
194        session_id: &str,
195        video_received: bool,
196        audio_received: bool,
197    ) -> Option<(bool, bool, Option<String>)> {
198        if let Some(entry) = self.mobile_peers.get_mut(session_id) {
199            if video_received {
200                entry.has_video = true;
201            }
202            if audio_received {
203                entry.has_audio = true;
204            }
205            return Some((
206                entry.has_video,
207                entry.has_audio,
208                entry.video_orientation.clone(),
209            ));
210        }
211        None
212    }
213
214    /// Inserts a new mobile peer. If a peer already exists for `session_id`,
215    /// it is replaced and returned.
216    pub fn insert_mobile_peer(
217        &mut self,
218        session_id: String,
219        device_id: String,
220        peer: Arc<RTCPeerConnection>,
221        video_orientation: Option<String>,
222    ) -> Option<Arc<RTCPeerConnection>> {
223        let previous = self.mobile_peers.insert(
224            session_id,
225            MobilePeer {
226                peer,
227                device_id,
228                has_video: false,
229                has_audio: false,
230                video_orientation,
231            },
232        );
233        previous.map(|entry| entry.peer)
234    }
235
236    /// Whether mobile preview video from `device_id` should be forwarded to
237    /// `mobile_preview` subscribers.
238    ///
239    /// * If [`Self::selected_mobile_preview_device`] is `Some(selected)`,
240    ///   only that device id is forwarded.
241    /// * If `None`, every mobile device is forwarded.
242    pub fn should_forward_mobile_preview(&self, device_id: &str) -> bool {
243        match self.selected_mobile_preview_device.as_deref() {
244            Some(selected) => selected == device_id,
245            None => true,
246        }
247    }

249    /// Ensures an RTP audio track exists for `device_id` and returns it.
250    ///
251    /// The track is created on first access with Opus codec parameters from
252    /// [`crate::streaming::signaling::opus_codec_capability`] and the label
253    /// `mobile_audio_{device_id}`.
254    pub fn ensure_device_audio_track(&mut self, device_id: &str) -> Arc<TrackLocalStaticRTP> {
255        self.device_audio_tracks
256            .entry(device_id.to_string())
257            .or_insert_with(|| {
258                Arc::new(TrackLocalStaticRTP::new(
259                    super::signaling::opus_codec_capability(),
260                    format!("mobile_audio_{device_id}"),
261                    "lumen".to_string(),
262                ))
263            })
264            .clone()
265    }
266
267    /// Whether there is at least one audio subscriber for `device_id`.
268    pub fn has_mobile_audio_subscriber(&self, device_id: &str) -> bool {
269        self.mobile_audio_peers
270            .get(device_id)
271            .map(|sessions| !sessions.is_empty())
272            .unwrap_or(false)
273    }
274
275    /// Inserts an audio-only peer for a specific device. Returns the
276    /// previous peer for that `(device_id, session_id)` if any.
277    pub fn insert_mobile_audio_peer(
278        &mut self,
279        device_id: &str,
280        session_id: &str,
281        peer: Arc<RTCPeerConnection>,
282    ) -> Option<Arc<RTCPeerConnection>> {
283        self.mobile_audio_peers
284            .entry(device_id.to_string())
285            .or_default()
286            .insert(session_id.to_string(), peer)
287    }
288
289    /// Returns the audio peer for `(device_id, session_id)` or `None` if not
290    /// found.
291    pub fn get_mobile_audio_peer(
292        &self,
293        device_id: &str,
294        session_id: &str,
295    ) -> Option<Arc<RTCPeerConnection>> {
296        self.mobile_audio_peers
297            .get(device_id)
298            .and_then(|sessions| sessions.get(session_id))
299            .cloned()
300    }
301
302    /// Removes an audio-only peer for `(device_id, session_id)`. If the
303    /// device's subscriber map becomes empty, the device entry is pruned.
304    pub fn remove_mobile_audio_peer(
305        &mut self,
306        device_id: &str,
307        session_id: &str,
308    ) -> Option<Arc<RTCPeerConnection>> {
309        let removed = self
310            .mobile_audio_peers
311            .get_mut(device_id)
312            .and_then(|sessions| sessions.remove(session_id));
313        if self
314            .mobile_audio_peers
315            .get(device_id)
316            .is_none_or(|sessions| sessions.is_empty())
317        {
318            self.mobile_audio_peers.remove(device_id);
319        }
320        removed
321    }
322
323    /// Removes all audio-only peers tied to `session_id` (from any device)
324    /// and returns the closed peer connections. Empty device maps are cleaned
325    /// up.
326    pub fn remove_mobile_audio_session(
327        &mut self,
328        session_id: &str,
329    ) -> Vec<Arc<RTCPeerConnection>> {
330        let mut closed = Vec::new();
331        let mut empty_devices = Vec::new();
332        for (device_id, sessions) in self.mobile_audio_peers.iter_mut() {
333            if let Some(peer) = sessions.remove(session_id) {
334                closed.push(peer);
335            }
336            if sessions.is_empty() {
337                empty_devices.push(device_id.clone());
338            }
339        }
340        for device_id in empty_devices {
341            self.mobile_audio_peers.remove(&device_id);
342        }
343        closed
344    }
345
346    /// Removes all peer connections owned by `session_id` across all stream
347    /// types.
348    ///
349    /// Returns a tuple:
350    /// `(preview, app_preview, main, mobile_preview, mobile)` where
351    /// `mobile` is `Some((peer, device_id))` if the session was a mobile
352    /// ingestion peer, else `None`.
353    pub fn remove_session(
354        &mut self,
355        session_id: &str,
356    ) -> (
357        Option<Arc<RTCPeerConnection>>,
358        Option<Arc<RTCPeerConnection>>,
359        Option<Arc<RTCPeerConnection>>,
360        Option<Arc<RTCPeerConnection>>,
361        Option<(Arc<RTCPeerConnection>, String)>,
362    ) {
363        let preview = self.preview_peers.remove(session_id);
364        let app_preview = self.app_preview_peers.remove(session_id);
365        let main = self.main_peers.remove(session_id);
366        let mobile_preview = self.mobile_preview_peers.remove(session_id);
367        let mobile = self
368            .mobile_peers
369            .remove(session_id)
370            .map(|entry| (entry.peer, entry.device_id));
371        (preview, app_preview, main, mobile_preview, mobile)
372    }
373}

375/// Initializes the streaming subsystem and registers its state with Tauri.
376///
377/// Boot sequence:
378/// 1. Ensures the `config` directory exists.
379/// 2. Creates the WebRTC [`MediaEngine`] and interceptor registry and builds
380///    the [`webrtc::api::API`].
381/// 3. Loads or creates the persisted [`StreamingConfig`] and starts the
382///    [`HtmlServerRuntime`] if enabled.
383/// 4. Creates all static local tracks (H264 for preview/app_preview/main and
384///    the RTP passthrough for mobile preview video).
385/// 5. Stores the [`StreamManager`] inside an [`Arc<Mutex<_>>`] wrapped by
386///    [`StreamingState`].
387/// 6. Starts the [`crate::streaming::app_preview_producer`] background task.
388/// 7. Emits the initial `streaming_status_changed` event.
389///
390/// # Errors
391///
392/// Returns the underlying error message if storage cannot be prepared, WebRTC
393/// setup fails, or the config cannot be loaded/written.
394pub fn initialize_streaming_state(app: &AppHandle) -> Result<StreamingState, String> {
395    ensure_streaming_storage()?;
396
397    let mut media_engine = MediaEngine::default();
398    media_engine
399        .register_default_codecs()
400        .map_err(|error| error.to_string())?;
401
402    let mut interceptor_registry = Registry::new();
403    interceptor_registry = register_default_interceptors(interceptor_registry, &mut media_engine)
404        .map_err(|error| error.to_string())?;
405
406    let api = APIBuilder::new()
407        .with_media_engine(media_engine)
408        .with_interceptor_registry(interceptor_registry)
409        .build();
410
411    let config = load_streaming_config()?;
412    let mut manager = StreamManager {
413        app: app.clone(),
414        api: Arc::new(api),
415        preview_track: Arc::new(TrackLocalStaticSample::new(
416            super::signaling::h264_codec_capability(),
417            "preview".to_string(),
418            "lumen".to_string(),
419        )),
420        app_preview_track: Arc::new(TrackLocalStaticSample::new(
421            super::signaling::h264_codec_capability(),
422            "app_preview".to_string(),
423            "lumen".to_string(),
424        )),
425        main_track: Arc::new(TrackLocalStaticSample::new(
426            super::signaling::h264_codec_capability(),
427            "main".to_string(),
428            "lumen".to_string(),
429        )),
430        mobile_preview_video_h264_track: Arc::new(TrackLocalStaticRTP::new(
431            super::signaling::h264_passthrough_codec_capability(),
432            "mobile_preview_video_h264".to_string(),
433            "lumen".to_string(),
434        )),
435        device_audio_tracks: HashMap::new(),
436        preview_peers: HashMap::new(),
437        app_preview_peers: HashMap::new(),
438        main_peers: HashMap::new(),
439        mobile_preview_peers: HashMap::new(),
440        mobile_peers: HashMap::new(),
441        mobile_audio_peers: HashMap::new(),
442        selected_mobile_preview_device: None,
443        is_content_protected: false,
444        html_server: HtmlServerRuntime::new(),
445        config,
446    };
447
448    manager.apply_html_server_config();
449    manager.emit_status();
450
451    let state = StreamingState {
452        manager: Arc::new(Mutex::new(manager)),
453    };
454
455    super::app_preview_producer::start_app_preview_producer(state.manager.clone());
456
457    Ok(state)
458}

475    Ok(state.manager.lock().await.config.clone())
476}

478/// Tauri command: updates and persists the streaming configuration.
479///
480/// **Frontend API:** `invoke('update_streaming_config', { config })`.
481///
482/// Behavior:
483/// 1. Runs [`sanitize_config`] on the provided value.
484/// 2. Persists the sanitized config via [`save_streaming_config`].
485/// 3. Applies it to the running [`StreamManager`]: restarts the HTML server
486///    if its enablement/port changed and recalculates
487///    `is_content_protected` (it remains true only if both the new config
488///    has `content_protection` enabled and the runtime flag was already
489///    true; use [`set_stream_content_protected`] to toggle runtime state).
490/// 4. Emits `streaming_status_changed`.
491///
492/// Returns the sanitized config that was stored/applied.
493///
494/// # Errors
495///
496/// Returns the underlying error message if the config cannot be saved.
497///
498/// # Examples
499///
500/// ```ts
501/// const updated = await invoke<StreamingConfig>('update_streaming_config', {
502///   config: nextCfg,
503/// });
504/// ```
505#[tauri::command]
506pub async fn update_streaming_config(
507    state: State<'_, StreamingState>,
508    config: StreamingConfig,
509) -> Result<StreamingConfig, String> {
510    let next = sanitize_config(config);
511    save_streaming_config(&next)?;
512
513    let mut manager = state.manager.lock().await;
514    let was_content_protected = manager.is_content_protected;
515    manager.config = next.clone();
516    manager.is_content_protected = manager.config.content_protection && was_content_protected;
517    manager.apply_html_server_config();
518    manager.emit_status();
519
520    Ok(next)
521}

634/// **Frontend API:** `invoke('push_stream_blank')`.
635///
636/// Broadcasts `{ "type": "blank" }` to all HTML presentation clients and
637/// updates the latest snapshot.
638#[tauri::command]
639pub async fn push_stream_blank(state: State<'_, StreamingState>) -> Result<(), String> {
640    let manager = state.manager.lock().await;

642    Ok(())

664/// Closes all peer connections associated with `session_id` and emits lifecycle
665/// events.
666///
667/// Called whenever a WebRTC session terminates (disconnect, fail, close) from
668/// the peer state change handlers, and by
669/// [`crate::streaming::signaling::handle_session_closed`]. It is an async
670/// helper that takes the `Arc<Mutex<StreamManager>>` directly (rather than
671/// `&mut self`) so it can drop the lock between lookups and `await`ing
672/// `peer.close()`.
673///
674/// Actions:
675/// 1. Extracts the app handle while holding the lock.
676/// 2. Removes every peer tied to `session_id` (all stream types plus mobile
677///    audio) and collects them.
678/// 3. Emits `streaming_status_changed`.
679/// 4. Closes all collected peers (preview/app_preview/main/mobile_preview
680///    and every mobile audio peer).
681/// 5. If a mobile ingestion peer was removed, closes it and emits
682///    `mobile_stream_ended` with `{ device_id }`.
683pub async fn cleanup_session(state: Arc<Mutex<StreamManager>>, session_id: &str) {
684    let app = {
685        let manager = state.lock().await;
686        manager.app.clone()
687    };
688
689    let ((preview, app_preview, main, mobile_preview, mobile), mobile_audio_peers) = {
690        let mut manager = state.lock().await;
691        let removed = manager.remove_session(session_id);
692        let mobile_audio_peers = manager.remove_mobile_audio_session(session_id);
693        manager.emit_status();
694        (removed, mobile_audio_peers)
695    };
696
697    if let Some(preview) = preview {
698        let _ = preview.close().await;
699    }
700
701    if let Some(app_preview) = app_preview {
702        let _ = app_preview.close().await;
703    }
704
705    if let Some(main) = main {
706        let _ = main.close().await;
707    }
708
709    if let Some(mobile_preview) = mobile_preview {
710        let _ = mobile_preview.close().await;
711    }
712
713    for mobile_audio_peer in mobile_audio_peers {
714        let _ = mobile_audio_peer.close().await;
715    }
716
717    if let Some((mobile, device_id)) = mobile {
718        let _ = mobile.close().await;
719        let _ = app.emit("mobile_stream_ended", json!({ "device_id": device_id }));
720    }
721}
