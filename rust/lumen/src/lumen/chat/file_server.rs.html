1//! Minimal localhost HTTP server for chat file attachments.
2//!
3//! Files saved by [`crate::chat::store::ChatStore::save_file`] live on disk
4//! under the chat files directory. This server exposes them at
5//! `http://127.0.0.1:{port}/files/{name}` so the desktop webview and remote
6//! devices can fetch attachments referenced by a [`crate::chat::store::ChatMessage`].
7//!
8//! The server binds to an ephemeral loopback port, accepts one request per
9//! connection, and only serves plain file names directly inside the files
10//! directory; anything else is rejected with `403` or `404`. Responses carry
11//! `Access-Control-Allow-Origin: *` so browsers can load them cross-origin.
12
13use std::path::PathBuf;
14
15use tokio::io::{AsyncReadExt, AsyncWriteExt};
16use tokio::net::TcpListener;
17
18/// Loopback HTTP server that serves chat attachment files.
19///
20/// Created once during [`crate::chat::initialize_chat_state`]; the bound port
21/// is read back with [`ChatFileServer::port`] and stored in
22/// [`crate::chat::ChatStateInner::file_server_port`].
23pub struct ChatFileServer {
24    /// Port assigned by the OS once [`ChatFileServer::start`] has bound the
25    /// listener; `0` before the server starts.
26    port: u16,
27    /// Directory whose files are exposed under the `/files/` route.
28    files_dir: PathBuf,
29}
30
31impl ChatFileServer {
32    /// Creates a server that will serve files from `files_dir`.
33    ///
34    /// The port is `0` until [`ChatFileServer::start`] is called.
35    pub fn new(files_dir: PathBuf) -> Self {
36        Self {
37            port: 0,
38            files_dir,
39        }
40    }
41
42    /// Returns the TCP port the server is listening on.
43    ///
44    /// Returns `0` if [`ChatFileServer::start`] has not run yet.
45    pub fn port(&self) -> u16 {
46        self.port
47    }
48
49    /// Binds to an ephemeral port on `127.0.0.1` and spawns the accept loop.
50    ///
51    /// Returns immediately after the listener is bound; connections are
52    /// handled on background tasks in the current Tokio runtime.
53    ///
54    /// # Errors
55    ///
56    /// Returns an error string when the socket cannot be bound or the local
57    /// address cannot be read.
58    pub async fn start(&mut self) -> Result<(), String> {
59        let listener = TcpListener::bind("127.0.0.1:0")
60            .await
61            .map_err(|e| e.to_string())?;
62        self.port = listener.local_addr().map_err(|e| e.to_string())?.port();
63
64        let files_dir = self.files_dir.clone();
65
66        tokio::spawn(async move {
67            loop {
68                let (stream, _addr) = match listener.accept().await {
69                    Ok(v) => v,
70                    Err(_) => continue,
71                };
72
73                let files_dir = files_dir.clone();
74                tokio::spawn(async move {
75                    if let Err(e) = handle_request(stream, &files_dir).await {
76                        eprintln!("[chat-file-server] error: {}", e);
77                    }
78                });
79            }
80        });
81
82        Ok(())
83    }
84}

86/// Serves a single HTTP request: `GET /files/{name}` from `files_dir`.
87///
88/// Requests for `/`, paths outside `/files/`, and file names containing path
89/// separators or `..` are rejected (`404` and `403` respectively).
90async fn handle_request(
91    mut stream: tokio::net::TcpStream,
92    files_dir: &PathBuf,
93) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
94    let mut buf = vec![0u8; 4096];
95    let n = stream.read(&mut buf).await?;
96    let request = String::from_utf8_lossy(&buf[..n]);
97
98    let path = request
99        .lines()
100        .next()
101        .and_then(|line| line.split_whitespace().nth(1))
102        .unwrap_or("/");
103
104    if path == "/" || !path.starts_with("/files/") {
105        let response = "HTTP/1.1 404 Not Found\r\nContent-Length: 0\r\n\r\n";
106        stream.write_all(response.as_bytes()).await?;
107        return Ok(());
108    }
109
110    let filename = &path[7..];
111    if filename.is_empty()
112        || filename.contains("..")
113        || filename.contains('\\')
114        || filename.contains('/')
115    {
116        let response = "HTTP/1.1 403 Forbidden\r\nContent-Length: 0\r\n\r\n";
117        stream.write_all(response.as_bytes()).await?;
118        return Ok(());
119    }
120
121    let file_path = files_dir.join(filename);
122    let mut file = match tokio::fs::File::open(&file_path).await {
123        Ok(f) => f,
124        Err(_) => {
125            let response = "HTTP/1.1 404 Not Found\r\nContent-Length: 0\r\n\r\n";
126            stream.write_all(response.as_bytes()).await?;
127            return Ok(());
128        }
129    };
130
131    let mut contents = Vec::new();
132    file.read_to_end(&mut contents).await?;
133
134    let mime = guess_mime(filename);
135    let response = format!(
136        "HTTP/1.1 200 OK\r\nContent-Type: {}\r\nContent-Length: {}\r\nAccess-Control-Allow-Origin: *\r\n\r\n",
137        mime,
138        contents.len()
139    );
140    stream.write_all(response.as_bytes()).await?;
141    stream.write_all(&contents).await?;
142
143    Ok(())
144}

146/// Maps a file extension to a `Content-Type` value.
147///
148/// Returns `application/octet-stream` for unknown or missing extensions.
149fn guess_mime(filename: &str) -> &'static str {
150    match filename.rsplit('.').next().unwrap_or("") {
151        "jpg" | "jpeg" => "image/jpeg",
152        "png" => "image/png",
153        "gif" => "image/gif",
154        "webp" => "image/webp",
155        "svg" => "image/svg+xml",
156        "pdf" => "application/pdf",
157        "mp4" => "video/mp4",
158        "mp3" => "audio/mpeg",
159        "txt" => "text/plain",
160        _ => "application/octet-stream",
161    }
162}
