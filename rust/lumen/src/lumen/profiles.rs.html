1//! User profiles: per-user appearance settings persisted as JSON.
2//!
3//! Each profile is one pretty-printed `<id>.json` file under
4//! `<base>/config/profiles/`, so profiles can be backed up, copied or synced
5//! by the user without going through the app. Listings skip files that do
6//! not parse instead of failing the whole read.
7
8use serde::{Deserialize, Serialize};
9
10use crate::remote;
11
12/// Creates the profile directory when missing and returns its path.
13fn profiles_dir() -> Result<std::path::PathBuf, String> {
14    let dir = remote::app_base_dir()?.join("config").join("profiles");
15    std::fs::create_dir_all(&dir).map_err(|e| e.to_string())?;
16    Ok(dir)
17}

19/// A saved user profile, stored as one JSON file per profile.
20#[derive(Debug, Clone, Serialize, Deserialize)]
21#[serde(rename_all = "camelCase")]
22pub struct Profile {
23    /// Stable identifier, also used as the profile's file name.
24    pub id: String,
25    /// Display name chosen by the user.
26    pub name: String,
27    /// UI locale pinned by the profile, when set.
28    pub language: Option<String>,
29    /// Color scheme, `"dark"` or `"light"`.
30    pub color_mode: String,
31    /// Accent color identifier from the theme presets.
32    pub accent_id: String,
33    /// Background used when a media item brings none of its own.
34    pub default_background: Option<DefaultBackground>,
35    /// Creation time in Unix milliseconds, used to order listings.
36    pub created_at: i64,
37}
38
39/// Reference to a background asset used when nothing else is set.
40#[derive(Debug, Clone, Serialize, Deserialize)]
41#[serde(rename_all = "camelCase")]
42pub struct DefaultBackground {
43    /// Kind of background (`"theme"`, `"image"` or `"video"`), serialized as `type`.
44    #[serde(rename = "type")]
45    pub kind: String,
46    /// Path or URL of the background asset.
47    pub src: String,
48    /// Display name of the asset.
49    pub name: String,
50}

52/// Returns every valid profile, oldest first.
53///
54/// Files that cannot be read abort the listing, while files that do not
55/// deserialize are skipped silently so one corrupt profile does not hide the
56/// rest.
57///
58/// # Errors
59///
60/// Returns an error when the profile directory cannot be created, cannot be
61/// listed, or one of the profile files cannot be read.
62#[tauri::command]
63pub async fn profile_list() -> Result<Vec<Profile>, String> {
64    let dir = profiles_dir()?;
65    if !dir.is_dir() {
66        return Ok(Vec::new());
67    }
68
69    let mut profiles: Vec<Profile> = Vec::new();
70
71    for entry in std::fs::read_dir(&dir).map_err(|e| e.to_string())?.flatten() {
72        let name = entry.file_name();
73        let name_str = name.to_string_lossy();
74        if !name_str.ends_with(".json") {
75            continue;
76        }
77
78        let path = entry.path();
79        let raw = std::fs::read_to_string(&path).map_err(|e| e.to_string())?;
80        match serde_json::from_str::<Profile>(&raw) {
81            Ok(profile) => profiles.push(profile),
82            Err(_) => {}
83        }
84    }
85
86    profiles.sort_by_key(|p| p.created_at);
87    Ok(profiles)
88}

90/// Persists a profile as `<id>.json`, creating or overwriting the file.
91///
92/// # Errors
93///
94/// Returns an error when the profile directory cannot be created, the
95/// profile cannot be serialized, or the file cannot be written.
96#[tauri::command]
97pub async fn profile_save(profile: Profile) -> Result<(), String> {
98    let dir = profiles_dir()?;
99    let path = dir.join(format!("{}.json", profile.id));
100    let json = serde_json::to_string_pretty(&profile).map_err(|e| e.to_string())?;
101    std::fs::write(&path, json).map_err(|e| e.to_string())?;
102    Ok(())
103}

105/// Deletes the profile file for `id`.
106///
107/// Removing a profile that no longer exists succeeds silently, so the
108/// frontend can enforce deletions without a prior existence check.
109///
110/// # Errors
111///
112/// Returns an error when the file exists but cannot be removed.
113#[tauri::command]
114pub async fn profile_delete(id: String) -> Result<(), String> {
115    let dir = profiles_dir()?;
116    let path = dir.join(format!("{id}.json"));
117    if path.exists() {
118        std::fs::remove_file(&path).map_err(|e| e.to_string())?;
119    }
120    Ok(())
121}
