32#[tauri::command]
33pub async fn profile_list() -> Result<Vec<Profile>, String> {
34    let dir = profiles_dir()?;
35    if !dir.is_dir() {
36        return Ok(Vec::new());
37    }
38
39    let mut profiles: Vec<Profile> = Vec::new();
40
41    for entry in std::fs::read_dir(&dir).map_err(|e| e.to_string())?.flatten() {
42        let name = entry.file_name();
43        let name_str = name.to_string_lossy();
44        if !name_str.ends_with(".json") {
45            continue;
46        }
47
48        let path = entry.path();
49        let raw = std::fs::read_to_string(&path).map_err(|e| e.to_string())?;
50        match serde_json::from_str::<Profile>(&raw) {
51            Ok(profile) => profiles.push(profile),
52            Err(_) => {}
53        }
54    }
55
56    profiles.sort_by_key(|p| p.created_at);
57    Ok(profiles)
58}

60#[tauri::command]
61pub async fn profile_save(profile: Profile) -> Result<(), String> {
62    let dir = profiles_dir()?;
63    let path = dir.join(format!("{}.json", profile.id));
64    let json = serde_json::to_string_pretty(&profile).map_err(|e| e.to_string())?;
65    std::fs::write(&path, json).map_err(|e| e.to_string())?;
66    Ok(())
67}
