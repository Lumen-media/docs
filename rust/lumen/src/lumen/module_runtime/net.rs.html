1//! Permission-checked HTTP proxy for modules.
2//!
3//! Modules cannot reach the network directly; they issue requests through the
4//! [`module_net_request`] Tauri command, which enforces a layered policy
5//! before any connection is opened:
6//!
7//! 1. only `https` URLs are accepted;
8//! 2. private, loopback, link-local, multicast and `.local`/`.localhost` hosts
9//!    are always blocked (see [`is_private_host`]);
10//! 3. when the manifest declares `permissions.network`, the URL must match one
11//!    of the patterns (see [`match_url_pattern`]);
12//! 4. hop-by-hop and security-sensitive request headers are rejected, request
13//!    bodies are limited to methods that allow them, and the response body is
14//!    capped by `max_bytes`.
15//!
16//! Responses are returned as a [`ModuleNetResponse`]; all failures use the
17//! stable, machine-readable [`ModuleNetError`] shape.
18

34/// HTTP method requested by a module.
35///
36/// Deserialized case-insensitively from an uppercase string (e.g. `"GET"`).
37/// When a request omits the method, [`module_net_request`] infers `POST` if a
38/// body is present and `GET` otherwise.
39#[derive(Debug, Deserialize, Clone)]
40#[serde(rename_all = "UPPERCASE")]
41pub enum NetMethod {
42    /// `GET`
43    Get,
44    /// `POST`
45    Post,
46    /// `PUT`
47    Put,
48    /// `PATCH`
49    Patch,
50    /// `DELETE`
51    Delete,
52    /// `HEAD`
53    Head,
54}
55
56impl NetMethod {
57    fn as_str(&self) -> &'static str {
58        match self {
59            NetMethod::Get => "GET",
60            NetMethod::Post => "POST",
61            NetMethod::Put => "PUT",
62            NetMethod::Patch => "PATCH",
63            NetMethod::Delete => "DELETE",
64            NetMethod::Head => "HEAD",
65        }
66    }
67
68    fn allows_body(&self) -> bool {
69        matches!(self, NetMethod::Post | NetMethod::Put | NetMethod::Patch)
70    }
71}

73/// How the response body should be decoded before being returned to the module.
74///
75/// Deserialized from a snake_case string; defaults to [`NetResponseType::Json`]
76/// when omitted.
77#[derive(Debug, Deserialize, Clone)]
78#[serde(rename_all = "snake_case")]
79pub enum NetResponseType {
80    /// Parse the body as JSON; falls back to the raw text string on parse error.
81    Json,
82    /// Return the body verbatim as a string.
83    Text,
84    /// Return the body as a base64-encoded string.
85    Bytes,
86    /// Discard the body; `data` is `null`.
87    None,
88}

90/// Request body variants for a module HTTP request.
91///
92/// Externally tagged by a `type` field, e.g. `{ "type": "json", "value": {...} }`.
93#[derive(Debug, Deserialize)]
94#[serde(tag = "type")]
95pub enum ModuleNetRequestBody {
96    /// JSON body, serialized with the `application/json` content type.
97    #[serde(rename = "json")]
98    Json { value: serde_json::Value },
99    /// Raw text body with an optional explicit content type.
100    #[serde(rename = "text")]
101    Text { value: String, #[serde(default)] content_type: Option<String> },
102    /// Binary body supplied as base64, with an optional explicit content type.
103    #[serde(rename = "bytes")]
104    Bytes { value_base64: String, #[serde(default)] content_type: Option<String> },
105    /// `application/x-www-form-urlencoded` body built from key/value pairs.
106    #[serde(rename = "form")]
107    Form { value: HashMap<String, String> },
108    /// `multipart/form-data` body. Reserved: currently rejected by
109    /// [`module_net_request`] as unsupported.
110    #[serde(rename = "multipart")]
111    Multipart { parts: Vec<MultipartPart> },
112}

137/// A network request issued by a module through [`module_net_request`].
138///
139/// Unknown fields are ignored and every field except `url` is optional.
140#[derive(Debug, Deserialize)]
141pub struct ModuleNetRequest {
142    /// Absolute target URL. Must be `https` and satisfy the module's
143    /// network permissions.
144    pub url: String,
145    /// HTTP method; inferred (`POST` with a body, otherwise `GET`) when absent.
146    #[serde(default)]
147    pub method: Option<NetMethod>,
148    /// Query parameters appended to `url`, each key allowing multiple values.
149    #[serde(default)]
150    pub query: Option<HashMap<String, Vec<String>>>,
151    /// Additional request headers. Hop-by-hop and `sec-*`/`proxy-*` headers are
152    /// rejected.
153    #[serde(default)]
154    pub headers: Option<HashMap<String, String>>,
155    /// Optional request body.
156    #[serde(default)]
157    pub body: Option<ModuleNetRequestBody>,
158    /// Desired response decoding; defaults to JSON.
159    #[serde(rename = "responseType", default)]
160    pub response_type: Option<NetResponseType>,
161    /// Per-request timeout in milliseconds, clamped to 60_000; default 15_000.
162    #[serde(rename = "timeoutMs", default)]
163    pub timeout_ms: Option<u64>,
164    /// Maximum response-body size in bytes, clamped to 50_000_000; default
165    /// 10_000_000. Exceeding it yields a `response_too_large` error.
166    #[serde(rename = "maxBytes", default)]
167    pub max_bytes: Option<u64>,
168    /// Whether to follow redirects. Currently informational: the shared HTTP
169    /// client follows up to 5 redirects regardless.
170    #[serde(rename = "followRedirects", default)]
171    pub follow_redirects: Option<bool>,
172}

174/// Successful (transport-level) result of a module HTTP request.
175///
176/// A non-2xx status is still a success here; inspect [`ModuleNetResponse::ok`]
177/// or [`ModuleNetResponse::status`].
178#[derive(Debug, Serialize, Clone)]
179pub struct ModuleNetResponse {
180    /// `true` when the status code is in the 2xx range.
181    pub ok: bool,
182    /// Numeric HTTP status code.
183    pub status: u16,
184    /// Canonical status reason phrase (`"Unknown"` if unavailable).
185    #[serde(rename = "statusText")]
186    pub status_text: String,
187    /// Response headers, with hop-by-hop/proxy headers filtered out.
188    pub headers: HashMap<String, String>,
189    /// Final URL after redirects.
190    pub url: String,
191    /// `true` when the final URL differs from the requested URL.
192    pub redirected: bool,
193    /// Decoded body according to the requested [`NetResponseType`]: a JSON
194    /// value, a string, a base64 string, or `null`.
195    pub data: serde_json::Value,
196}

226/// Returns `true` for hosts that must never be reachable by a module.
227///
228/// Covers `localhost`, `127.0.0.1`, `::1`, `.local`/`.localhost` names, and
229/// loopback/private/link-local/unspecified/multicast IP literals (SSRF guard).
230fn is_private_host(host: &str) -> bool {
231    if host == "localhost" || host == "127.0.0.1" || host == "::1" {
232        return true;
233    }
234    if host.ends_with(".local") || host.ends_with(".localhost") {
235        return true;
236    }
237    if let Ok(ip) = host.parse::<std::net::IpAddr>() {
238        match ip {
239            std::net::IpAddr::V4(v4) => {
240                return v4.is_loopback()
241                    || v4.is_private()
242                    || v4.is_link_local()
243                    || v4.is_unspecified()
244                    || v4.is_multicast();
245            }
246            std::net::IpAddr::V6(v6) => {
247                return v6.is_loopback() || v6.is_unspecified() || v6.is_multicast();
248            }
249        }
250    }
251    false
252}

254/// Matches `url_str` against a manifest network pattern.
255///
256/// Schemes and hosts must match; a `*.example.com` pattern matches any
257/// subdomain but not the bare domain. A pattern path ending in `*` acts as a
258/// prefix match, otherwise paths must be equal. Unparseable inputs never match.
259fn match_url_pattern(url_str: &str, pattern: &str) -> bool {
260    fn inner(url_str: &str, pattern: &str) -> Option<bool> {
261        let url = Url::parse(url_str).ok()?;
262        let pattern_url = Url::parse(pattern).ok()?;
263
264        if url.scheme() != pattern_url.scheme() {
265            return Some(false);
266        }
267
268        let url_host = url.host_str().unwrap_or("");
269        let pattern_host = pattern_url.host_str().unwrap_or("");
270
271        if pattern_host.starts_with("*.") {
272            let suffix = &pattern_host[1..];
273            if !url_host.ends_with(suffix) {
274                return Some(false);
275            }
276            if url_host == &suffix[1..] {
277                return Some(false);
278            }
279        } else if url_host != pattern_host {
280            return Some(false);
281        }
282
283        let url_path = url.path();
284        let pattern_path = pattern_url.path();
285
286        if pattern_path.ends_with('*') {
287            let prefix = &pattern_path[..pattern_path.len() - 1];
288            if !url_path.starts_with(prefix) {
289                return Some(false);
290            }
291        } else if url_path != pattern_path {
292            return Some(false);
293        }
294
295        Some(true)
296    }
297    inner(url_str, pattern).unwrap_or(false)
298}

300/// Enforces the URL policy for `url_str` against `manifest`.
301///
302/// Rejects non-`https` URLs, private/loopback hosts, and URLs that fail to
303/// match a non-empty `permissions.network` allow-list. When the manifest
304/// declares no network permissions, any public HTTPS host is permitted.
305///
306/// # Errors
307///
308/// Returns a [`ModuleNetError`] with code `invalid_url`, `blocked_url` or
309/// `permission_denied` depending on the violated rule. Also used by the themes
310/// module before downloading remote images.
311pub(crate) fn check_url_allowed(
312    url_str: &str,
313    manifest: &ModuleManifest,
314) -> Result<(), ModuleNetError> {
315    let url = Url::parse(url_str).map_err(|_| ModuleNetError {
316        code: "invalid_url".into(),
317        message: format!("invalid URL: {url_str}"),
318        status: None,
319        url: Some(url_str.into()),
320    })?;
321
322    if url.scheme() != "https" {
323        return Err(ModuleNetError {
324            code: "blocked_url".into(),
325            message: "only https URLs are allowed".into(),
326            status: None,
327            url: Some(url_str.into()),
328        });
329    }
330
331    if let Some(host) = url.host_str() {
332        if is_private_host(host) {
333            return Err(ModuleNetError {
334                code: "blocked_url".into(),
335                message: format!("private network host blocked: {host}"),
336                status: None,
337                url: Some(url_str.into()),
338            });
339        }
340    }
341
342    if let Some(permissions) = &manifest.permissions {
343        if !permissions.network.is_empty() {
344            let allowed = permissions
345                .network
346                .iter()
347                .any(|p| match_url_pattern(url_str, p));
348            if !allowed {
349                return Err(ModuleNetError {
350                    code: "permission_denied".into(),
351                    message: format!("URL not allowed by module permissions: {url_str}"),
352                    status: None,
353                    url: Some(url_str.into()),
354                });
355            }
356        }
357    }
358
359    Ok(())
360}

364/// Appends `query` parameters onto `base_url` and returns the resulting URL.
365fn build_url(mut base_url: Url, query: Option<HashMap<String, Vec<String>>>) -> Url {
366    if let Some(query_params) = query {
367        {
368            let mut pairs = base_url.query_pairs_mut();
369            for (key, values) in query_params {
370                for value in values {
371                    pairs.append_pair(&key, &value);
372                }
373            }
374        }
375    }
376    base_url
377}

379/// Translates a [`ModuleNetRequest`] into a configured [`reqwest::RequestBuilder`].
380///
381/// Resolves the method, applies query parameters, validates and attaches
382/// headers (rejecting forbidden ones), encodes the body according to its
383/// variant, and sets the (clamped) timeout.
384///
385/// # Errors
386///
387/// Returns a [`ModuleNetError`] for invalid methods/URLs/headers, forbidden
388/// headers, a body on a method that does not allow one, or unsupported body
389/// kinds (multipart).
390fn build_request(
391    client: &Client,
392    input: &ModuleNetRequest,
393) -> Result<reqwest::RequestBuilder, ModuleNetError> {
394    let method_str = match &input.method {
395        Some(m) => m.as_str(),
396        None => {
397            if input.body.is_some() {
398                "POST"
399            } else {
400                "GET"
401            }
402        }
403    };
404
405    let method: reqwest::Method = method_str.parse().map_err(|_| ModuleNetError {
406        code: "network_error".into(),
407        message: format!("invalid HTTP method: {method_str}"),
408        status: None,
409        url: Some(input.url.clone()),
410    })?;
411
412    let allows_body = input
413        .method
414        .as_ref()
415        .map(|m| m.allows_body())
416        .unwrap_or(input.body.is_some());
417
418    let mut url = Url::parse(&input.url).map_err(|_| ModuleNetError {
419        code: "invalid_url".into(),
420        message: format!("invalid URL: {}", input.url),
421        status: None,
422        url: Some(input.url.clone()),
423    })?;
424
425    url = build_url(url, input.query.clone());
426
427    let mut req = client.request(method, url.clone());
428
429    if let Some(headers) = &input.headers {
430        let mut header_map = reqwest::header::HeaderMap::new();
431        for (key, value) in headers {
432            let lower = key.to_lowercase();
433            if matches!(
434                lower.as_str(),
435                "host" | "content-length" | "connection" | "transfer-encoding" | "upgrade"
436            ) || lower.starts_with("proxy-")
437                || lower.starts_with("sec-")
438            {
439                return Err(ModuleNetError {
440                    code: "permission_denied".into(),
441                    message: format!("header not allowed: {key}"),
442                    status: None,
443                    url: Some(url.to_string()),
444                });
445            }
446
447            let header_name = key.parse::<reqwest::header::HeaderName>().map_err(|_| {
448                ModuleNetError {
449                    code: "network_error".into(),
450                    message: format!("invalid header name: {key}"),
451                    status: None,
452                    url: Some(url.to_string()),
453                }
454            })?;
455
456            if let Ok(header_value) = value.parse::<reqwest::header::HeaderValue>() {
457                header_map.insert(header_name, header_value);
458            }
459        }
460        req = req.headers(header_map);
461    }
462
463    if let Some(body) = &input.body {
464        if !allows_body {
465            return Err(ModuleNetError {
466                code: "unsupported_body".into(),
467                message: format!("request body not allowed for {} requests", method_str),
468                status: None,
469                url: Some(url.to_string()),
470            });
471        }
472
473        match body {
474            ModuleNetRequestBody::Json { value } => {
475                req = req.json(value);
476            }
477            ModuleNetRequestBody::Text {
478                value,
479                content_type,
480            } => {
481                req = req.body(value.clone());
482                if let Some(ct) = content_type {
483                    req = req.header(reqwest::header::CONTENT_TYPE, ct.clone());
484                }
485            }
486            ModuleNetRequestBody::Bytes {
487                value_base64,
488                content_type,
489            } => {
490                let bytes = base64::engine::general_purpose::STANDARD
491                    .decode(value_base64)
492                    .map_err(|_| ModuleNetError {
493                        code: "unsupported_body".into(),
494                        message: "invalid base64 in request body".into(),
495                        status: None,
496                        url: Some(url.to_string()),
497                    })?;
498                req = req.body(bytes);
499                if let Some(ct) = content_type {
500                    req = req.header(reqwest::header::CONTENT_TYPE, ct.clone());
501                }
502            }
503            ModuleNetRequestBody::Form { value } => {
504                let pairs: Vec<(String, String)> = value.clone().into_iter().collect();
505                req = req.form(&pairs);
506            }
507            ModuleNetRequestBody::Multipart { parts: _ } => {
508                return Err(ModuleNetError {
509                    code: "unsupported_body".into(),
510                    message: "multipart body not yet supported".into(),
511                    status: None,
512                    url: Some(url.to_string()),
513                });
514            }
515        };
516    }
517
518    let timeout_ms = input.timeout_ms.unwrap_or(15_000).min(60_000);
519    req = req.timeout(Duration::from_millis(timeout_ms));
520
521    Ok(req)
522}

524// ── Tauri command ─────────────────────────────────────────────────

526/// Tauri command: performs a permission-checked HTTP request for a module.
527///
528/// # Frontend API
529///
530/// ```text
531/// invoke("module_net_request", { moduleId, input })
532/// ```
533///
534/// `module_id` must be a registered, loadable module; its manifest
535/// `permissions.network` patterns gate the target URL. The request is executed
536/// with the shared [`crate::module_runtime::ModuleRuntime`] HTTP client and the
537/// decoded result is returned as a [`ModuleNetResponse`].
538///
539/// # Errors
540///
541/// Returns a [`ModuleNetError`] when the module is unknown, its manifest cannot
542/// be loaded, the URL fails `check_url_allowed`, the request cannot be built,
543/// the transport fails/times out, or the body exceeds `max_bytes`.
544#[tauri::command]
545pub async fn module_net_request(
546    app: AppHandle,
547    module_id: String,
548    input: ModuleNetRequest,
549) -> Result<ModuleNetResponse, ModuleNetError> {
550    let runtime = app.state::<ModuleRuntime>();
551
552    let entry = {
553        let reg = runtime.registry.lock().map_err(|e: PoisonError<_>| ModuleNetError {
554            code: "network_error".into(),
555            message: e.to_string(),
556            status: None,
557            url: Some(input.url.clone()),
558        })?;
559
560        reg.get(&module_id)
561            .map_err(|e: rusqlite::Error| ModuleNetError {
562                code: "network_error".into(),
563                message: e.to_string(),
564                status: None,
565                url: Some(input.url.clone()),
566            })?
567            .ok_or_else(|| ModuleNetError {
568                code: "permission_denied".into(),
569                message: format!("module not found: {module_id}"),
570                status: None,
571                url: Some(input.url.clone()),
572            })?
573    };
574
575    let manifest = super::manifest::load_manifest(&entry.path).map_err(|e| ModuleNetError {
576        code: "network_error".into(),
577        message: format!("failed to load module manifest: {e}"),
578        status: None,
579        url: Some(input.url.clone()),
580    })?;
581
582    check_url_allowed(&input.url, &manifest)?;
583
584    let client = &runtime.http_client;
585    let max_bytes = input.max_bytes.unwrap_or(10_000_000).min(50_000_000);
586    let response_type = input.response_type.clone().unwrap_or(NetResponseType::Json);
587
588    let request_builder = build_request(client, &input)?;
589    let response = request_builder.send().await.map_err(|e| {
590        if e.is_timeout() {
591            ModuleNetError {
592                code: "timeout".into(),
593                message: "request timed out".into(),
594                status: None,
595                url: None,
596            }
597        } else if e.is_connect() {
598            ModuleNetError {
599                code: "network_error".into(),
600                message: format!("connection failed: {e}"),
601                status: None,
602                url: None,
603            }
604        } else {
605            ModuleNetError {
606                code: "network_error".into(),
607                message: format!("request failed: {e}"),
608                status: None,
609                url: None,
610            }
611        }
612    })?;
613
614    let status = response.status();
615    let status_text = status.canonical_reason().unwrap_or("Unknown").to_string();
616    let response_url = response.url().to_string();
617    let redirected = response.url().as_str() != input.url;
618
619    let headers: HashMap<String, String> = response
620        .headers()
621        .iter()
622        .filter(|(k, _)| {
623            let lower = k.as_str().to_lowercase();
624            !matches!(
625                lower.as_str(),
626                "transfer-encoding"
627                    | "connection"
628                    | "keep-alive"
629                    | "proxy-authenticate"
630                    | "proxy-authorization"
631                    | "te"
632                    | "trailer"
633                    | "upgrade"
634            ) && !lower.starts_with("proxy-")
635        })
636        .map(|(k, v)| (k.as_str().to_string(), v.to_str().unwrap_or("").to_string()))
637        .collect();
638
639    let data = read_response_body(response, &response_type, max_bytes, &response_url).await?;
640
641    Ok(ModuleNetResponse {
642        ok: status.is_success(),
643        status: status.as_u16(),
644        status_text,
645        headers,
646        url: response_url,
647        redirected,
648        data,
649    })
650}

652/// Reads and decodes a response body according to `response_type`.
653///
654/// Enforces the `max_bytes` cap before returning; binary bodies are returned as
655/// base64 strings. JSON parse failures fall back to the raw text.
656///
657/// # Errors
658///
659/// Returns a [`ModuleNetError`] with code `invalid_response` when the body
660/// cannot be read, or `response_too_large` when it exceeds `max_bytes`.
661async fn read_response_body(
662    response: reqwest::Response,
663    response_type: &NetResponseType,
664    max_bytes: u64,
665    response_url: &str,
666) -> Result<serde_json::Value, ModuleNetError> {
667    match response_type {
668        NetResponseType::Json | NetResponseType::Text => {
669            let text = response.text().await.map_err(|e| ModuleNetError {
670                code: "invalid_response".into(),
671                message: format!("failed to read response body: {e}"),
672                status: None,
673                url: Some(response_url.into()),
674            })?;
675
676            if text.len() > max_bytes as usize {
677                return Err(ModuleNetError {
678                    code: "response_too_large".into(),
679                    message: format!("response exceeds {} byte limit", max_bytes),
680                    status: None,
681                    url: Some(response_url.into()),
682                });
683            }
684
685            match response_type {
686                NetResponseType::Json => {
687                    Ok(serde_json::from_str(&text).unwrap_or(serde_json::Value::String(text)))
688                }
689                NetResponseType::Text => Ok(serde_json::Value::String(text)),
690                _ => unreachable!(),
691            }
692        }
693        NetResponseType::Bytes => {
694            let bytes = response.bytes().await.map_err(|e| ModuleNetError {
695                code: "invalid_response".into(),
696                message: format!("failed to read response body: {e}"),
697                status: None,
698                url: Some(response_url.into()),
699            })?;
700
701            if bytes.len() as u64 > max_bytes {
702                return Err(ModuleNetError {
703                    code: "response_too_large".into(),
704                    message: format!("response exceeds {} byte limit", max_bytes),
705                    status: None,
706                    url: Some(response_url.into()),
707                });
708            }
709
710            Ok(serde_json::Value::String(
711                base64::engine::general_purpose::STANDARD.encode(&bytes),
712            ))
713        }
714        NetResponseType::None => Ok(serde_json::Value::Null),
715    }
716}
