1//! Theme image imports for modules.
2//!
3//! Modules use backgrounds and other imagery through the shared `theme_files`
4//! table in the main Lumen database. [`module_theme_add`] lets a module import
5//! an image either from a URL (subject to the same network permissions as
6//! [`crate::module_runtime::net`]) or from a path inside its own directory, then
7//! copies it into the `themes` media directory and records a row.
8//!
9//! Imports are content-hashed with BLAKE3: re-importing the exact same bytes
10//! returns the existing row instead of writing a duplicate. Filenames are
11//! sanitized and uniquified on disk. Imported files are served back to the
12//! webview through the `lumen-module://__theme/id/{id}` route (see
13//! [`crate::module_runtime::protocol`]).
14

26/// File extensions accepted as theme images (without a leading dot).
27const IMAGE_EXTS: &[&str] = &["gif", "jpg", "jpeg", "png", "webp", "svg", "bmp", "avif"];
28/// Maximum size of an image that may be imported (50 MB).
29const MAX_IMAGE_BYTES: usize = 50_000_000;
30
31/// Where a module theme image is imported from.
32///
33/// Externally tagged by a `type` field: `{"type":"url","url":...}` or
34/// `{"type":"file","path":...}`.
35#[derive(Debug, Deserialize)]
36#[serde(tag = "type", rename_all = "snake_case")]
37pub enum ModuleThemeSource {
38    /// Download the image from a remote URL (checked against the module's
39    /// network permissions).
40    Url { url: String },
41    /// Read the image from a path inside the module directory (path-traversal
42    /// protected).
43    File { path: String },
44}
45
46/// Input for [`module_theme_add`].
47#[derive(Debug, Deserialize)]
48pub struct ModuleThemeAddInput {
49    /// The image source.
50    pub source: ModuleThemeSource,
51    /// Optional preferred base name for the stored file.
52    #[serde(default)]
53    pub name: Option<String>,
54}
55
56/// Result of [`module_theme_add`], describing the stored image.
57#[derive(Debug, Serialize)]
58pub struct ModuleThemeAddResult {
59    /// Row id in the `theme_files` table, usable in `lumen-module://__theme/id/{id}`.
60    pub id: i64,
61    /// Final file name on disk (without directory).
62    pub name: String,
63    /// Absolute path to the stored image file.
64    pub path: String,
65    /// Lowercase file extension including the leading dot.
66    pub extension: String,
67}

69/// Tauri command: imports an image into the shared themes store for a module.
70///
71/// # Frontend API
72///
73/// ```text
74/// invoke("module_theme_add", { moduleId, input })
75/// ```
76///
77/// The module must exist and be enabled. For URL sources the download is
78/// subject to the module's network permissions and a size cap; for file sources
79/// the path is scoped to the module directory. The image is content-hashed so
80/// duplicate imports return the existing entry.
81///
82/// # Errors
83///
84/// Returns an error when the module is missing/disabled, its manifest cannot be
85/// loaded, the source is unauthorized or outside the module directory, the
86/// download fails, the image type is unsupported, the size cap is exceeded, or
87/// any filesystem/database operation fails.
88#[tauri::command]
89pub async fn module_theme_add(
90    app: AppHandle,
91    module_id: String,
92    input: ModuleThemeAddInput,
93) -> Result<ModuleThemeAddResult, String> {
94    let runtime = app.state::<ModuleRuntime>();
95
96    let entry = runtime
97        .registry
98        .lock()
99        .map_err(|e| e.to_string())?
100        .get(&module_id)
101        .map_err(|e| e.to_string())?
102        .ok_or_else(|| format!("module not found: {module_id}"))?;
103
104    if !entry.enabled {
105        return Err("module is not enabled".into());
106    }
107
108    let manifest = load_manifest(&entry.path)?;
109
110    let (bytes, ext, stem) = match &input.source {
111        ModuleThemeSource::Url { url } => {
112            check_url_allowed(url, &manifest).map_err(|e| e.to_string())?;
113
114            let response = runtime
115                .http_client
116                .get(url)
117                .send()
118                .await
119                .map_err(|e| format!("failed to download image: {e}"))?;
120
121            if !response.status().is_success() {
122                return Err(format!("image download failed: HTTP {}", response.status()));
123            }
124
125            let content_type = response
126                .headers()
127                .get(reqwest::header::CONTENT_TYPE)
128                .and_then(|value| value.to_str().ok())
129                .map(|value| value.to_string());
130
131            let bytes = response.bytes().await.map_err(|e| e.to_string())?;
132            if bytes.len() > MAX_IMAGE_BYTES {
133                return Err(format!("image exceeds {MAX_IMAGE_BYTES} byte limit"));
134            }
135
136            let ext = mime_to_ext(content_type.as_deref())
137                .or_else(|| url_extension(url).filter(|ext| is_image_ext(ext)))
138                .unwrap_or_else(|| ".jpg".to_string());
139
140            (bytes.to_vec(), ext, url_stem(url))
141        }
142        ModuleThemeSource::File { path } => {
143            let full = scoped_module_path(&app, &module_id, path)?;
144            let bytes = std::fs::read(&full).map_err(|e| format!("failed to read file: {e}"))?;
145
146            let file_name = Path::new(path)
147                .file_name()
148                .map(|s| s.to_string_lossy().to_string())
149                .unwrap_or_default();
150
151            let (stem, ext) = split_name(&file_name);
152            (bytes, ext, stem)
153        }
154    };
155
156    let ext = ext.to_ascii_lowercase();
157    if !is_image_ext(&ext) {
158        return Err(format!("unsupported image type: {ext}"));
159    }
160
161    let db_path = crate::paths::db_path()?;
162    let connection = Connection::open(&db_path).map_err(|e| e.to_string())?;
163
164    let content_hash = blake3::hash(&bytes).to_hex().to_string();
165
166    // Dedup by content: if this exact image already exists, return it instead
167    // of writing a new file + row. Prevents duplicates from rapid double calls.
168    let existing = connection
169        .query_row(
170            "SELECT id, name, path, extension FROM theme_files WHERE content_hash = ?1",
171            params![content_hash],
172            |row| {
173                Ok(ModuleThemeAddResult {
174                    id: row.get(0)?,
175                    name: row.get(1)?,
176                    path: row.get(2)?,
177                    extension: row.get(3)?,
178                })
179            },
180        )
181        .ok();
182
183    if let Some(result) = existing {
184        return Ok(result);
185    }
186
187    let themes_dir = crate::paths::media_dir("themes")?;
188    std::fs::create_dir_all(&themes_dir).map_err(|e| e.to_string())?;
189
190    let mut desired = sanitize_filename(input.name.as_deref().unwrap_or(&stem));
191    if desired.is_empty() {
192        desired = "background".to_string();
193    }
194
195    let (file_path, file_name) = unique_destination(&themes_dir, &desired, &ext);
196    std::fs::write(&file_path, &bytes).map_err(|e| e.to_string())?;
197
198    let path_str = file_path.to_string_lossy().to_string();
199    let size = std::fs::metadata(&file_path)
200        .map_err(|e| e.to_string())?
201        .len() as i64;
202    let modified_at = SystemTime::now()
203        .duration_since(UNIX_EPOCH)
204        .map_err(|e| e.to_string())?
205        .as_millis() as i64;
206
207    connection
208        .execute(
209            "INSERT OR IGNORE INTO theme_files (name, path, size, modified_at, extension, content_hash)
210             VALUES (?1, ?2, ?3, ?4, ?5, ?6)",
211            params![file_name, path_str, size, modified_at, ext, content_hash],
212        )
213        .map_err(|e| e.to_string())?;
214
215    let id: i64 = connection
216        .query_row(
217            "SELECT id FROM theme_files WHERE path = ?1",
218            params![path_str],
219            |row| row.get(0),
220        )
221        .map_err(|e| e.to_string())?;
222
223    Ok(ModuleThemeAddResult {
224        id,
225        name: file_name,
226        path: path_str,
227        extension: ext,
228    })
229}

231/// Returns `true` if `ext` (with or without a leading dot) is an image type.
232fn is_image_ext(ext: &str) -> bool {
233    let ext = ext.trim_start_matches('.').to_ascii_lowercase();
234    IMAGE_EXTS.contains(&ext.as_str())
235}

237/// Maps an HTTP `Content-Type` to a file extension, ignoring parameters.
238fn mime_to_ext(content_type: Option<&str>) -> Option<String> {
239    let ct = content_type?.split(';').next()?.trim().to_ascii_lowercase();
240    let ext = match ct.as_str() {
241        "image/jpeg" => ".jpg",
242        "image/png" => ".png",
243        "image/gif" => ".gif",
244        "image/webp" => ".webp",
245        "image/svg+xml" => ".svg",
246        "image/bmp" => ".bmp",
247        "image/avif" => ".avif",
248        _ => return None,
249    };
250    Some(ext.to_string())
251}

253/// Extracts the lowercased extension (including the dot) from a URL path.
254fn url_extension(url: &str) -> Option<String> {
255    let parsed = url::Url::parse(url).ok()?;
256    let name = parsed.path().rsplit('/').next()?;
257    let dot = name.rfind('.')?;
258    let ext = &name[dot + 1..];
259    if ext.is_empty() {
260        None
261    } else {
262        Some(format!(".{ext}"))
263    }
264}

266/// Derives a base file name (without extension) from a URL path.
267///
268/// Falls back to `"background"` when the URL has no usable name.
269fn url_stem(url: &str) -> String {
270    let parsed = url::Url::parse(url).ok();
271    let path = parsed.as_ref().map(|u| u.path()).unwrap_or(url);
272    let name = path.rsplit('/').next().unwrap_or("");
273    let stem = match name.rfind('.') {
274        Some(dot) => &name[..dot],
275        None => name,
276    };
277    if stem.is_empty() {
278        "background".to_string()
279    } else {
280        stem.to_string()
281    }
282}

284/// Splits a file name into `(stem, extension)`, lowercasing the extension.
285fn split_name(name: &str) -> (String, String) {
286    match name.rfind('.') {
287        Some(dot) if dot > 0 => (name[..dot].to_string(), name[dot..].to_ascii_lowercase()),
288        _ => (name.to_string(), String::new()),
289    }
290}

292/// Replaces filesystem-unsafe characters in `name` with `-` and trims trailing
293/// dots/spaces. May return an empty string, which callers substitute.
294fn sanitize_filename(name: &str) -> String {
295    let mut out = String::new();
296    for ch in name.trim().chars() {
297        if ch.is_control() || matches!(ch, '<' | '>' | ':' | '"' | '/' | '\\' | '|' | '?' | '*') {
298            out.push('-');
299        } else {
300            out.push(ch);
301        }
302    }
303    while out.ends_with('.') || out.ends_with(' ') {
304        out.pop();
305    }
306    out
307}

309/// Finds a non-colliding `(path, file_name)` in `dir` by appending ` (n)`.
310fn unique_destination(dir: &Path, name: &str, ext: &str) -> (PathBuf, String) {
311    let mut candidate = format!("{name}{ext}");
312    let mut n = 1;
313    while dir.join(&candidate).exists() {
314        candidate = format!("{name} ({n}){ext}");
315        n += 1;
316    }
317    (dir.join(&candidate), candidate)
318}
