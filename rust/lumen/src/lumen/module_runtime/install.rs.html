1//! Module installation from `.lumenpack` archives or directories.
2//!
3//! [`Installer`] copies a module into the modules root and registers it. Two
4//! sources are supported:
5//!
6//! * a `.lumenpack` file, a ZIP archive containing `manifest.json` plus the
7//!   module payload, extracted after validating every entry path;
8//! * a directory, either copied into the modules root (sideload) or referenced
9//!   in place (`dev_mode`, so the developer's build output is served live).
10//!
11//! Both extraction routines treat the archive and the manifest id as untrusted
12//! input. [`resolve_module_dest`] and [`resolve_archive_entry`] reject absolute
13//! paths, `..`/`~` segments and any path that would escape the destination
14//! directory (Zip Slip), and every rejection is covered by the unit tests in
15//! this file.
16

25/// Performs the physical install of a module and records it in the [`Registry`].
26///
27/// Borrows the modules root and a registry for the duration of the install;
28/// typically constructed ad hoc inside a Tauri command (see
29/// [`crate::module_runtime::module_install`] and
30/// [`crate::module_runtime::store::store_install`]).
31///
32/// # Examples
33///
34/// ```no_run
35/// # use std::path::Path;
36/// # use crate::module_runtime::install::Installer;
37/// # use crate::module_runtime::registry::Registry;
38/// let registry = Registry::open(Path::new("/data/lumen.sqlite"))?;
39/// let installer = Installer::new(Path::new("/data/modules"), &registry);
40/// let manifest = installer.install_from_path(Path::new("/tmp/demo.lumenpack"), false)?;
41/// println!("installed {}", manifest.id);
42/// # Ok::<(), Box<dyn std::error::Error>>(())
43/// ```
44pub struct Installer<'a> {
45    modules_dir: &'a Path,
46    registry: &'a Registry,
47}
48
49impl<'a> Installer<'a> {
50    /// Creates an installer targeting `modules_dir` and writing to `registry`.
51    pub fn new(modules_dir: &'a Path, registry: &'a Registry) -> Self {
52        Self { modules_dir, registry }
53    }
54
55    /// Installs a module from `source_path` and returns its manifest.
56    ///
57    /// Dispatches on the source: a `.lumenpack` extension installs from the
58    /// ZIP archive; a directory installs via [`Installer::install_from_dir`];
59    /// anything else is rejected. On success the module is inserted into the
60    /// registry with `enabled = true`.
61    ///
62    /// # Examples
63    ///
64    /// ```no_run
65    /// # use std::path::Path;
66    /// # use crate::module_runtime::install::Installer;
67    /// # use crate::module_runtime::registry::Registry;
68    /// # let registry = Registry::open(Path::new("/data/lumen.sqlite"))?;
69    /// # let installer = Installer::new(Path::new("/data/modules"), &registry);
70    /// // Sideload a pack, or install a directory in dev mode to serve it live.
71    /// let manifest = installer.install_from_path(Path::new("/src/my-module"), true)?;
72    /// # Ok::<(), Box<dyn std::error::Error>>(())
73    /// ```
74    ///
75    /// # Errors
76    ///
77    /// Returns an error string when the source is neither a `.lumenpack` file
78    /// nor a directory, when the manifest is missing/invalid, when the module
79    /// id is unsafe to use as a path segment, when an archive entry would
80    /// escape the destination, or on any filesystem failure.
81    pub fn install_from_path(
82        &self,
83        source_path: &Path,
84        dev_mode: bool,
85    ) -> Result<ModuleManifest, String> {
86        let ext = source_path
87            .extension()
88            .and_then(|e| e.to_str())
89            .unwrap_or("");
90
91        if ext == "lumenpack" {
92            self.install_from_pack(source_path)
93        } else if source_path.is_dir() {
94            self.install_from_dir(source_path, dev_mode)
95        } else {
96            Err(format!(
97                "unsupported source: {}. Expected a .lumenpack file or a directory.",
98                source_path.display()
99            ))
100        }
101    }
102
103    /// Extracts a `.lumenpack` ZIP archive into the modules root and registers it.
104    ///
105    /// Reads and validates `manifest.json` from the archive, removes any
106    /// pre-existing install of the same id, then extracts every entry via
107    /// [`resolve_archive_entry`]. The recorded `source` is `"sideload"`.
108    fn install_from_pack(&self, pack_path: &Path) -> Result<ModuleManifest, String> {
109        let file = fs::File::open(pack_path)
110            .map_err(|e| format!("cannot open pack: {e}"))?;
111        let mut archive =
112            ZipArchive::new(file).map_err(|e| format!("invalid zip archive: {e}"))?;
113
114        let manifest = extract_manifest_from_archive(&mut archive)?;
115        manifest.validate()?;
116
117        let dest = resolve_module_dest(self.modules_dir, &manifest.id)?;
118        if dest.exists() {
119            fs::remove_dir_all(&dest).map_err(|e| format!("cannot remove existing dir: {e}"))?;
120        }
121        fs::create_dir_all(&dest).map_err(|e| format!("cannot create module dir: {e}"))?;
122
123        for i in 0..archive.len() {
124            let mut file = archive.by_index(i).map_err(|e| e.to_string())?;
125            let entry_name = file.name().to_string();
126            let out_path = resolve_archive_entry(&dest, &entry_name)?;
127
128            if entry_name.replace('\\', "/").ends_with('/') {
129                fs::create_dir_all(&out_path).map_err(|e| e.to_string())?;
130            } else {
131                if let Some(parent) = out_path.parent() {
132                    fs::create_dir_all(parent).map_err(|e| e.to_string())?;
133                }
134                let mut out =
135                    fs::File::create(&out_path).map_err(|e| e.to_string())?;
136                std::io::copy(&mut file, &mut out).map_err(|e| e.to_string())?;
137            }
138        }
139
140        self.registry
141            .insert(&ModuleEntry {
142                id: manifest.id.clone(),
143                version: manifest.version.clone(),
144                source: "sideload".into(),
145                enabled: true,
146                path: dest,
147            })
148            .map_err(|e| e.to_string())?;
149
150        Ok(manifest)
151    }

153    /// Installs a module from an on-disk directory.
154    ///
155    /// The manifest is loaded from `dir`; in `dev_mode` the directory is
156    /// registered in place (`source = "dev"`, changes served live), otherwise
157    /// it is copied into the modules root (`source = "sideload"`).
158    fn install_from_dir(&self, dir: &Path, dev_mode: bool) -> Result<ModuleManifest, String> {
159        let manifest = load_manifest(dir)?;
160
161        let dest: PathBuf = if dev_mode {
162            dir.to_path_buf()
163        } else {
164            let d = resolve_module_dest(self.modules_dir, &manifest.id)?;
165            if d.exists() {
166                fs::remove_dir_all(&d).map_err(|e| format!("cannot remove existing dir: {e}"))?;
167            }
168            copy_dir_all(dir, &d)?;
169            d
170        };
171
172        self.registry
173            .insert(&ModuleEntry {
174                id: manifest.id.clone(),
175                version: manifest.version.clone(),
176                source: if dev_mode { "dev" } else { "sideload" }.into(),
177                enabled: true,
178                path: dest,
179            })
180            .map_err(|e| e.to_string())?;
181
182        Ok(manifest)
183    }
184}

186/// Resolves and validates the destination directory for `module_id`.
187///
188/// Rejects empty ids, ids containing path separators, `.`/`..`, absolute
189/// paths, and any join that would not stay strictly inside `modules_dir`.
190/// This is the guard that prevents a hostile manifest id from escaping the
191/// modules root.
192fn resolve_module_dest(modules_dir: &Path, module_id: &str) -> Result<PathBuf, String> {
193    let reject = || format!("invalid module id: {module_id}");
194
195    if module_id.is_empty() {
196        return Err(reject());
197    }
198    if module_id.contains('/') || module_id.contains('\\') {
199        return Err(reject());
200    }
201    if module_id == "." || module_id == ".." {
202        return Err(reject());
203    }
204    if Path::new(module_id).is_absolute() {
205        return Err(reject());
206    }
207
208    let dest = modules_dir.join(module_id);
209    if !dest.starts_with(modules_dir) || dest == modules_dir {
210        return Err(reject());
211    }
212
213    Ok(dest)
214}

216/// Resolves a ZIP entry name to a safe path under `base`.
217///
218/// Normalizes Windows separators, then rejects absolute paths and any `..`/`~`
219/// segment. The final join is re-checked to stay inside `base`, blocking Zip
220/// Slip attacks from crafted archives.
221fn resolve_archive_entry(base: &Path, entry_name: &str) -> Result<PathBuf, String> {
222    let candidate = entry_name.replace('\\', "/");
223    let reject = || format!("archive entry escapes module directory: {entry_name}");
224
225    if candidate.starts_with('/') || Path::new(&candidate).is_absolute() {
226        return Err(reject());
227    }
228
229    if candidate
230        .split('/')
231        .any(|segment| segment == ".." || segment == "~")
232    {
233        return Err(reject());
234    }
235
236    let out_path = base.join(&candidate);
237    if !out_path.starts_with(base) {
238        return Err(reject());
239    }
240
241    Ok(out_path)
242}

244/// Reads and parses `manifest.json` from the root of a pack archive.
245///
246/// Does not run [`ModuleManifest::validate`]; callers do that after extraction
247/// so that the id can be inspected first.
248fn extract_manifest_from_archive(
249    archive: &mut ZipArchive<fs::File>,
250) -> Result<ModuleManifest, String> {
251    let mut file = archive
252        .by_name("manifest.json")
253        .map_err(|_| "manifest.json not found in archive")?;
254    let mut content = String::new();
255    file.read_to_string(&mut content)
256        .map_err(|e| format!("cannot read manifest.json: {e}"))?;
257    serde_json::from_str(&content).map_err(|e| format!("invalid manifest.json: {e}"))
258}
259
260/// Recursively copies the directory tree at `src` into `dst`.
261fn copy_dir_all(src: &Path, dst: &Path) -> Result<(), String> {
262    fs::create_dir_all(dst).map_err(|e| e.to_string())?;
263    for entry in fs::read_dir(src).map_err(|e| e.to_string())? {
264        let entry = entry.map_err(|e| e.to_string())?;
265        let dst_path = dst.join(entry.file_name());
266        if entry.file_type().map_err(|e| e.to_string())?.is_dir() {
267            copy_dir_all(&entry.path(), &dst_path)?;
268        } else {
269            fs::copy(entry.path(), &dst_path).map_err(|e| e.to_string())?;
270        }
271    }
272    Ok(())
273}

343    fn write_pack(dir: &Path, id: &str, entries: &[(&str, &str)]) -> PathBuf {
344        use std::io::Write;
345        let pack = dir.join("test.lumenpack");
346        let file = fs::File::create(&pack).unwrap();
347        let mut zip = zip::ZipWriter::new(file);
348        let manifest = format!(
349            r#"{{"id":"{id}","name":"T","version":"1.0.0","api":"1"}}"#
350        );
351        zip.start_file("manifest.json", zip::write::SimpleFileOptions::default())
352            .unwrap();
353        zip.write_all(manifest.as_bytes()).unwrap();
354        for (name, body) in entries {
355            zip.start_file(*name, zip::write::SimpleFileOptions::default())
356                .unwrap();
357            zip.write_all(body.as_bytes()).unwrap();
358        }
359        zip.finish().unwrap();
360        pack
361    }
