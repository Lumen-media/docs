20pub fn handle_lumen_request(
21    ctx: UriSchemeContext<'_, tauri::Wry>,
22    request: tauri::http::Request<Vec<u8>>,
23    responder: UriSchemeResponder,
24) {
25    let uri = request.uri().to_string();
26    let is_thumb = uri.contains("lumen-thumb");
27
28    let path_part = if is_thumb {
29        uri.strip_prefix("lumen-thumb://")
30            .or_else(|| uri.strip_prefix("http://lumen-thumb.localhost/"))
31            .or_else(|| uri.strip_prefix("https://lumen-thumb.localhost/"))
32    } else {
33        uri.strip_prefix("lumen://")
34            .or_else(|| uri.strip_prefix("http://lumen.localhost/"))
35            .or_else(|| uri.strip_prefix("https://lumen.localhost/"))
36    }
37    .unwrap_or("")
38    .trim_start_matches('/');
39
40    let (_, query) = match path_part.split_once('?') {
41        Some(p) => p,
42        None => ("", ""),
43    };
44
45    let mut src = String::new();
46    let mut req_w: Option<u32> = None;
47    let mut req_q: Option<u8> = None;
48    for (k, v) in form_urlencoded::parse(query.as_bytes()) {
49        match k.as_ref() {
50            "src" => src = v.into_owned(),
51            "w" => req_w = v.parse().ok(),
52            "q" => req_q = v.parse().ok(),
53            _ => {}
54        }
55    }
56
57    if src.is_empty() {
58        responder.respond(response(400, "Bad Request: missing src parameter"));
59        return;
60    }
61
62    let cache_root = match ctx
63        .app_handle()
64        .path()
65        .app_data_dir()
66        .map(|d| d.join("lumen").join("cache"))
67    {
68        Ok(d) => d,
69        Err(e) => {
70            responder.respond(response(500, e.to_string()));
71            return;
72        }
73    };
74
75    let app = ctx.app_handle().clone();
76    let emit_src = src.clone();
77
78    std::thread::spawn(move || {
79        let _ = app.emit("lumen:optimizing", &emit_src);
80        let res = process(&cache_root, &src, is_thumb, req_w, req_q).map(|(bytes, mime)| {
81            tauri::http::Response::builder()
82                .status(200)
83                .header("Content-Type", mime)
84                .header("Access-Control-Allow-Origin", "*")
85                .header("Cache-Control", "public, max-age=31536000, immutable")
86                .body(bytes)
87                .unwrap()
88        });
89        let _ = app.emit("lumen:optimized", &emit_src);
90        responder.respond(res.unwrap_or_else(|(status, msg)| response(status, msg)));
91    });
92}

94fn process(
95    cache_root: &Path,
96    src: &str,
97    is_thumb: bool,
98    req_w: Option<u32>,
99    req_q: Option<u8>,
100) -> Result<(Vec<u8>, &'static str), (u16, String)> {
101    let is_remote = src.starts_with("http://") || src.starts_with("https://");
102
103    let has_dims = req_w.is_some();
104
105    if has_dims || is_thumb {
106        return sized(&local_cache_dir(cache_root, is_remote), src, is_remote, req_w, req_q)
107            .map(|bytes| (bytes, "image/jpeg"));
108    }
109
110    if is_remote {
111        if is_unsplash(src) {
112            let dir = local_cache_dir(cache_root, true);
113            std::fs::create_dir_all(&dir).map_err(io_err)?;
114            let hash = blake3::hash(src.as_bytes()).to_hex();
115            let dest = dir.join(format!("{hash}_full.jpg"));
116            if let Ok(cached) = std::fs::read(&dest) {
117                return Ok((cached, "image/jpeg"));
118            }
119            let bytes = fetch(&unsplash_optimized(src, None, None, JPEG_QUALITY_FULL))?;
120            let _ = std::fs::write(&dest, &bytes);
121            return Ok((bytes, "image/jpeg"));
122        }
123        let bytes = fetch(src)?;
124        match image::load_from_memory(&bytes) {
125            Ok(img) => {
126                let dir = local_cache_dir(cache_root, true);
127                std::fs::create_dir_all(&dir).map_err(io_err)?;
128                let hash = blake3::hash(src.as_bytes()).to_hex();
129                let dest = dir.join(format!("{hash}_full.jpg"));
130                if let Ok(cached) = std::fs::read(&dest) {
131                    return Ok((cached, "image/jpeg"));
132                }
133                write_jpeg(&img, &dest, JPEG_QUALITY_FULL).map_err(str_err)?;
134                std::fs::read(&dest)
135                    .map(|b| (b, "image/jpeg"))
136                    .map_err(io_err)
137            }
138            Err(_) => Ok((bytes, mime_for_url(src))),
139        }
140    } else {
141        let path = Path::new(src);
142        if !path.exists() {
143            return Err((404, format!("file not found: {src}")));
144        }
145        let ext = extension_of(src);
146        if is_image_ext(&ext) {
147            let dir = local_cache_dir(cache_root, false);
148            std::fs::create_dir_all(&dir).map_err(io_err)?;
149            let hash = blake3::hash(src.as_bytes()).to_hex();
150            let dest = dir.join(format!("{hash}_full.jpg"));
151            if let Ok(cached) = std::fs::read(&dest) {
152                return Ok((cached, "image/jpeg"));
153            }
154            let img = image::open(path).map_err(|e| (400, format!("image decode: {e}")))?;
155            write_jpeg(&img, &dest, JPEG_QUALITY_FULL).map_err(str_err)?;
156            std::fs::read(&dest)
157                .map(|b| (b, "image/jpeg"))
158                .map_err(io_err)
159        } else if is_video_ext(&ext) {
160            Ok((std::fs::read(path).map_err(io_err)?, mime_for_ext(&ext)))
161        } else {
162            Err((415, format!("unsupported source for lumen://: {src}")))
163        }
164    }
165}

167fn sized(
168    dir: &Path,
169    src: &str,
170    is_remote: bool,
171    req_w: Option<u32>,
172    req_q: Option<u8>,
173) -> Result<Vec<u8>, (u16, String)> {
174    std::fs::create_dir_all(dir).map_err(io_err)?;
175
176    let hash = blake3::hash(src.as_bytes()).to_hex();
177    let w = req_w.map(|v| v.max(1)).unwrap_or(DEFAULT_THUMB);
178    let q = req_q.unwrap_or(JPEG_QUALITY_THUMB).clamp(1, 100);
179
180    if let Some(dest) = find_cached(dir, &hash, w, q) {
181        return std::fs::read(&dest).map_err(io_err);
182    }
183
184    let path = Path::new(src);
185    if !path.exists() {
186        return Err((404, format!("file not found: {src}")));
187    }
188
189    let cache_name = |w: u32, h: u32| format!("{hash}_q{q}_{w}x{h}.jpg");
190
191    if is_remote {
192        if is_unsplash(src) {
193            let bytes = fetch(&unsplash_optimized(src, Some(w), None, q))?;
194            let (aw, ah) = image_dimensions_bytes(&bytes).map_err(|e| (400, e))?;
195            let dest = dir.join(cache_name(aw, ah));
196            std::fs::write(&dest, &bytes).map_err(io_err)?;
197            cache_insert(dir, &hash, aw, ah, q, dest.clone());
198            return Ok(bytes);
199        }
200        let bytes = fetch(src)?;
201        let img = image::load_from_memory(&bytes).map_err(|e| (400, format!("image decode: {e}")))?;
202        let (ow, oh) = (img.width(), img.height());
203        let h = derive_h(w, ow, oh);
204        let dest = dir.join(cache_name(w, h));
205        write_jpeg(&img.thumbnail(w, h), &dest, q).map_err(str_err)?;
206        cache_insert(dir, &hash, w, h, q, dest.clone());
207        return std::fs::read(&dest).map_err(io_err);
208    }
209
210    let ext = extension_of(src);
211    if is_image_ext(&ext) {
212        let img = decode_local(path, w).ok_or_else(|| {
213            (400, format!("image decode: {src}"))
214        })?;
215        let (ow, oh) = (img.width(), img.height());
216        let h = derive_h(w, ow, oh);
217        let dest = dir.join(cache_name(w, h));
218        write_jpeg(&img.thumbnail(w, h), &dest, q).map_err(str_err)?;
219        cache_insert(dir, &hash, w, h, q, dest.clone());
220        std::fs::read(&dest).map_err(io_err)
221    } else if is_video_ext(&ext) {
222        let (vw, vh, dest) = if let Some(img) = os_thumb::get_thumbnail(path, w) {
223            let (vw, vh) = (img.width(), img.height());
224            let dest = dir.join(cache_name(vw, vh));
225            write_jpeg(&img, &dest, q).map_err(str_err)?;
226            (vw, vh, dest)
227        } else {
228            let tmp = dir.join(format!("{hash}_q{q}_w{w}.jpg"));
229            let (vw, vh) = video_thumb::generate_box_width(path, &tmp, w, q).map_err(str_err)?;
230            let dest = dir.join(cache_name(vw, vh));
231            if dest != tmp {
232                std::fs::rename(&tmp, &dest).map_err(io_err)?;
233            }
234            (vw, vh, dest)
235        };
236        cache_insert(dir, &hash, vw, vh, q, dest.clone());
237        std::fs::read(&dest).map_err(io_err)
238    } else {
239        Err((415, format!("unsupported source for lumen-thumb://: {src}")))
240    }
241}

243fn derive_h(w: u32, ow: u32, oh: u32) -> u32 {
244    if ow == 0 {
245        return w;
246    }
247    ((w as u64 * oh as u64) / ow as u64).max(1) as u32
248}

250/// Fast path for local images: try the OS thumbnail first (Shell cache), then a
251/// scaled JPEG decode that only decodes the DCT blocks needed for the target
252/// width, falling back to a full `image` decode.
253fn decode_local(path: &Path, w: u32) -> Option<image::DynamicImage> {
254    if let Some(img) = os_thumb::get_thumbnail(path, w) {
255        return Some(img);
256    }
257    if let Some(img) = decode_scaled_jpeg(path, w) {
258        return Some(img);
259    }
260    image::open(path).ok()
261}

263/// Decode a JPEG at a reduced scale so we never materialize the full image.
264fn decode_scaled_jpeg(path: &Path, w: u32) -> Option<image::DynamicImage> {
265    let file = std::fs::File::open(path).ok()?;
266    let mut decoder = jpeg_decoder::Decoder::new(std::io::BufReader::new(file));
267    decoder.read_info().ok()?;
268    let info = decoder.info()?;
269    let ow = info.width as u32;
270    let oh = info.height as u32;
271    if ow == 0 || oh == 0 {
272        return None;
273    }
274    let h = derive_h(w, ow, oh);
275    let (sw, sh) = decoder.scale(w.min(u16::MAX as u32) as u16, h.min(u16::MAX as u32) as u16).ok()?;
276    let pixels = decoder.decode().ok()?;
277    let img = image::RgbImage::from_raw(sw as u32, sh as u32, pixels)?;
278    Some(image::DynamicImage::ImageRgb8(img))
279}

281fn image_dimensions_bytes(bytes: &[u8]) -> Result<(u32, u32), String> {
282    let img = image::load_from_memory(bytes).map_err(|e| format!("image decode: {e}"))?;
283    Ok((img.width(), img.height()))

286fn find_cached(dir: &Path, hash: &str, w: u32, q: u8) -> Option<PathBuf> {
287    let index_key = format!("{}::{}", dir.display(), hash);
288    let index = INDEX.get_or_init(|| Mutex::new(HashMap::new()));
289    let mut lock = index.lock().unwrap();
290    if !lock.contains_key(&index_key) {
291        lock.insert(index_key.clone(), scan_cache(dir, hash));
292    }
293    let entries = &lock[&index_key];
294
295    let max_w = (w as f64 * TOLERANCE).ceil() as u64;
296
297    let mut best: Option<(u64, PathBuf)> = None;
298    for (cw, ch, cq, path) in entries {
299        if *cq == q && *cw >= w && (*cw as u64) <= max_w {
300            let area = (*cw as u64) * (*ch as u64);
301            if best.as_ref().map_or(true, |(a, _)| area < *a) {
302                best = Some((area, path.clone()));
303            }
304        }
305    }
306    best.map(|(_, p)| p)
307}

309fn scan_cache(dir: &Path, hash: &str) -> Vec<(u32, u32, u8, PathBuf)> {
310    let prefix = format!("{hash}_");
311    let mut out = Vec::new();
312    if let Ok(entries) = std::fs::read_dir(dir) {
313        for entry in entries.flatten() {
314            let name = entry.file_name().to_string_lossy().into_owned();
315            if let Some(rest) = name.strip_prefix(&prefix) {
316                if let Some((w, h, q)) = parse_dims(rest) {
317                    out.push((w, h, q, entry.path()));
318                }
319            }
320        }
321    }
322    out
323}

325fn parse_dims(rest: &str) -> Option<(u32, u32, u8)> {
326    let rest = rest.strip_suffix(".jpg")?;
327    let (qpart, dims) = rest.split_once('_')?;
328    let q = qpart.strip_prefix('q')?.parse().ok()?;
329    let (w, h) = dims.split_once('x')?;
330    Some((w.parse().ok()?, h.parse().ok()?, q))
331}

333fn cache_insert(dir: &Path, hash: &str, w: u32, h: u32, q: u8, path: PathBuf) {
334    let index_key = format!("{}::{}", dir.display(), hash);
335    let index = INDEX.get_or_init(|| Mutex::new(HashMap::new()));
336    let mut lock = index.lock().unwrap();
337    lock.entry(index_key).or_default().push((w, h, q, path));
338}

340fn write_jpeg(img: &image::DynamicImage, dest: &Path, quality: u8) -> Result<(), String> {
341    let file = std::fs::File::create(dest).map_err(|e| format!("create cache file: {e}"))?;
342    let mut encoder = image::codecs::jpeg::JpegEncoder::new_with_quality(file, quality);
343    encoder
344        .encode_image(&img.to_rgb8())
345        .map_err(|e| format!("encode jpeg: {e}"))
346}

348fn fetch(url: &str) -> Result<Vec<u8>, (u16, String)> {
349    let resp = reqwest::blocking::get(url).map_err(|e| (502, format!("fetch {url}: {e}")))?;
350    if !resp.status().is_success() {
351        return Err((502, format!("upstream status {} for {url}", resp.status())));
352    }
353    resp.bytes()
354        .map(|b| b.to_vec())
355        .map_err(|e| (502, format!("read {url}: {e}")))
356}

358fn is_unsplash(src: &str) -> bool {
359    Url::parse(src)
360        .ok()
361        .and_then(|u| u.host_str().map(str::to_ascii_lowercase))
362        .map(|h| h == "images.unsplash.com" || h.ends_with(".unsplash.com"))
363        .unwrap_or(false)
364}

366fn unsplash_optimized(src: &str, w: Option<u32>, h: Option<u32>, q: u8) -> String {
367    let mut url = match Url::parse(src) {
368        Ok(u) => u,
369        Err(_) => return src.to_string(),
370    };
371    {
372        let mut pairs = url.query_pairs_mut();
373        if let Some(w) = w {
374            pairs.append_pair("w", &w.to_string());
375        }
376        if let Some(h) = h {
377            pairs.append_pair("h", &h.to_string());
378        }
379        pairs.append_pair("q", &q.to_string());
380        pairs.append_pair(
381            "fit",
382            if w.is_some() && h.is_some() { "crop" } else { "max" },
383        );
384        pairs.append_pair("fm", "jpg");
385        pairs.append_pair("auto", "format");
386    }
387    url.to_string()
388}

390fn is_image_ext(ext: &str) -> bool {
391    matches!(ext, "png" | "jpg" | "jpeg" | "webp" | "bmp" | "gif")
392}
393
394fn is_video_ext(ext: &str) -> bool {
395    matches!(ext, "mp4" | "mov" | "m4v" | "webm" | "avi" | "mkv")
396}

398fn extension_of(src: &str) -> String {
399    Path::new(src)
400        .extension()
401        .and_then(|e| e.to_str())
402        .map(|e| e.to_lowercase())
403        .unwrap_or_default()
404}

406fn mime_for_url(url: &str) -> &'static str {
407    let path = url.split('?').next().unwrap_or(url);
408    let ext = extension_of(path);
409    if is_video_ext(&ext) {
410        mime_for_ext(&ext)
411    } else {
412        "application/octet-stream"
413    }
414}
415
416fn mime_for_ext(ext: &str) -> &'static str {
417    match ext {
418        "png" => "image/png",
419        "jpg" | "jpeg" => "image/jpeg",
420        "gif" => "image/gif",
421        "webp" => "image/webp",
422        "bmp" => "image/bmp",
423        "mp4" => "video/mp4",
424        "webm" => "video/webm",
425        "mov" => "video/quicktime",
426        "avi" => "video/x-msvideo",
427        "mkv" => "video/x-matroska",
428        _ => "application/octet-stream",
429    }
430}
