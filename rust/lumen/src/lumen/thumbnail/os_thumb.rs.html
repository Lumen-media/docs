1//! Operating-system thumbnail providers.
2//!
3//! Requests a native thumbnail from the platform instead of decoding the file
4//! ourselves, which is dramatically cheaper for videos and large images:
5//!
6//! - **Windows**: `IShellItemImageFactory` (Shell thumbnail cache) converted
7//!   from `HBITMAP` to BGRA pixels.
8//! - **Linux**: the FreeDesktop thumbnail cache spec
9//!   (`~/.cache/thumbnails/{normal,large}/<md5 of file:// URI>.png`).
10//! - **macOS / other**: not supported, always returns `None` so callers fall
11//!   back to [`super::image_thumb`] or [`super::video_thumb`].
12//!
13//! This module is the first resolution step of [`super::get_thumbnail`] and of
14//! the fast path in [`super::protocol`].
15
16use std::path::Path;
17
18/// Tries to save an OS-provided thumbnail of `src` into `dest` as JPEG.
19///
20/// The image returned by the platform is fitted into a `size` x `size` box
21/// before encoding. Returns `true` only when the thumbnail was produced and
22/// written successfully; any missing provider, decode failure, or I/O error
23/// results in `false`.
24pub fn try_get(src: &Path, dest: &Path, size: u32) -> bool {
25    if let Some(img) = get_thumbnail(src, size) {
26        let img = img.thumbnail(size, size);
27        img.save_with_format(dest, image::ImageFormat::Jpeg).is_ok()
28    } else {
29        false
30    }
31}
32
33/// Returns the platform's native thumbnail for `src`, if one is available.
34///
35/// The image is returned unscaled beyond what the provider itself produced
36/// (`size` is only a hint passed to the OS). Returns `None` when the platform
37/// has no thumbnail for the file, when the provider is unsupported (macOS and
38/// other targets), or when reading the cached thumbnail fails.
39pub fn get_thumbnail(src: &Path, size: u32) -> Option<image::DynamicImage> {
40    platform::get_thumbnail(src, size)
41}

43// ── Windows: IShellItemImageFactory → HBITMAP → BGRA pixels ──────────────────
44
45#[cfg(target_os = "windows")]
46mod platform {
47    use std::path::Path;
48
49    /// Windows provider backed by the Shell thumbnail cache.
50    pub fn get_thumbnail(src: &Path, size: u32) -> Option<image::DynamicImage> {
51        unsafe { via_shell(src, size) }
52    }
53
54    /// Requests a `size` x `size` thumbnail through
55    /// `IShellItemImageFactory::GetImage` with `SIIGBF_THUMBNAILONLY`, reads the
56    /// resulting `HBITMAP` into a top-down 32-bit DIB, swaps the BGRA channels
57    /// into RGBA, and wraps the pixels in a [`image::DynamicImage`].
58    ///
59    /// All GDI objects created here (compatible DC, selected bitmap) are
60    /// released on every path, including early returns.
61    ///
62    /// # Safety
63    ///
64    /// Calls Win32 GDI/COM APIs with raw handles and pointers. The caller must
65    /// ensure the current thread can initialize COM (best effort here via
66    /// `CoInitializeEx`) and that `src` is a path that can be lossily converted
67    /// to a wide string; the returned image is only produced when every GDI
68    /// call succeeds, otherwise `None` is returned.
69    unsafe fn via_shell(src: &Path, size: u32) -> Option<image::DynamicImage> {
70        use windows::Win32::Foundation::SIZE;
71        use windows::Win32::Graphics::Gdi::{
72            CreateCompatibleDC, DeleteDC, GetDC, GetDIBits, GetObjectW, ReleaseDC, SelectObject,
73            BITMAP, BITMAPINFO, BITMAPINFOHEADER, BI_RGB, DIB_RGB_COLORS, HGDIOBJ,
74        };
75        use windows::Win32::System::Com::{CoInitializeEx, COINIT};
76        use windows::Win32::UI::Shell::{IShellItemImageFactory, SHCreateItemFromParsingName, SIIGBF};
77        use windows::core::PCWSTR;
78
79        const FLAGS: SIIGBF = SIIGBF(0x01);
80
81        let _ = CoInitializeEx(None, COINIT(0x2));
82
83        let normalized = src.to_string_lossy().replace('/', "\\");
84        let wide: Vec<u16> = normalized
85            .encode_utf16()
86            .chain(std::iter::once(0))
87            .collect();
88
89        let factory: IShellItemImageFactory =
90            SHCreateItemFromParsingName(PCWSTR(wide.as_ptr()), None).ok()?;
91
92        let sz = SIZE { cx: size as i32, cy: size as i32 };
93        let hbitmap = factory.GetImage(sz, FLAGS).ok()?;
94
95        let mut bm = BITMAP::default();
96        GetObjectW(
97            HGDIOBJ(hbitmap.0),
98            std::mem::size_of::<BITMAP>() as i32,
99            Some(&mut bm as *mut BITMAP as *mut _),
100        );
101
102        let w = bm.bmWidth;
103        let h = bm.bmHeight.abs();
104
105        if w == 0 || h == 0 {
106            windows::Win32::Graphics::Gdi::DeleteObject(HGDIOBJ(hbitmap.0));
107            return None;
108        }
109
110        let mut bmi = BITMAPINFO {
111            bmiHeader: BITMAPINFOHEADER {
112                biSize: std::mem::size_of::<BITMAPINFOHEADER>() as u32,
113                biWidth: w,
114                biHeight: -h, // negative = top-down scanlines
115                biPlanes: 1,
116                biBitCount: 32,
117                biCompression: BI_RGB.0,
118                ..Default::default()
119            },
120            ..Default::default()
121        };
122
123        let mut pixels = vec![0u8; (w * h * 4) as usize];
124        let hdc_screen = GetDC(None);
125        let hdc_mem = CreateCompatibleDC(hdc_screen);
126        let old_bmp = SelectObject(hdc_mem, HGDIOBJ(hbitmap.0));
127
128        let lines = GetDIBits(
129            hdc_mem,
130            hbitmap,
131            0,
132            h as u32,
133            Some(pixels.as_mut_ptr() as *mut _),
134            &mut bmi,
135            DIB_RGB_COLORS,
136        );
137
138        SelectObject(hdc_mem, old_bmp);
139        DeleteDC(hdc_mem);
140        ReleaseDC(None, hdc_screen);
141        windows::Win32::Graphics::Gdi::DeleteObject(HGDIOBJ(hbitmap.0));
142
143        if lines == 0 {
144            return None;
145        }
146
147        for chunk in pixels.chunks_exact_mut(4) {
148            chunk.swap(0, 2);
149        }
150
151        let img = image::RgbaImage::from_raw(w as u32, h as u32, pixels)?;
152        Some(image::DynamicImage::ImageRgba8(img))
153    }
154}

156// ── Linux: Freedesktop thumbnail spec (md5 of file:// URI) ───────────────────
157
158/// Linux provider reading the FreeDesktop thumbnail cache.
159///
160/// Looks up `<md5("file://" + path)>.png` under `~/.cache/thumbnails/normal`
161/// (requests up to 128 px) or `.../large` (above 128 px) and opens it if it
162/// exists; `None` means the cache has no entry yet (no thumbnail daemon ran,
163/// or the file was never indexed).
164#[cfg(target_os = "linux")]
165mod platform {
166    use std::path::Path;
167
168    /// Returns the cached FreeDesktop thumbnail for `src`, if present.
169    pub fn get_thumbnail(src: &Path, size: u32) -> Option<image::DynamicImage> {
170        // filename = md5(file://path).png; normal/ = 128px max, large/ = 256px max
171        let uri = format!("file://{}", src.to_string_lossy());
172        let hash = format!("{:x}", md5::compute(uri.as_bytes()));
173
174        let subdir = if size <= 128 { "normal" } else { "large" };
175
176        let cache = std::env::var("HOME")
177            .map(|h| std::path::PathBuf::from(h).join(".cache").join("thumbnails"))
178            .ok()?;
179
180        let thumb = cache.join(subdir).join(format!("{hash}.png"));
181
182        if !thumb.exists() {
183            return None;
184        }
185
186        image::open(&thumb).ok()
187    }
188}

190// ── macOS: not supported (QuickLook cache is proprietary) ────────────────────
191
192/// macOS provider stub.
193///
194/// QuickLook's thumbnail cache is a proprietary format with no public API, so
195/// there is nothing to read here; callers always fall back to decoding.
196#[cfg(target_os = "macos")]
197mod platform {
198    use std::path::Path;
199
200    /// Always returns `None` (no supported OS thumbnail source on macOS).
201    pub fn get_thumbnail(_src: &Path, _size: u32) -> Option<image::DynamicImage> {
202        None
203    }
204}
