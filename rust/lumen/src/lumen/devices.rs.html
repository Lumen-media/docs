186pub fn resolve_desktop_name(app: &AppHandle) -> String {
187    for key in ["COMPUTERNAME", "HOSTNAME"] {
188        if let Ok(value) = std::env::var(key) {
189            let trimmed = value.trim();
190            if !trimmed.is_empty() {
191                return trimmed.to_string();
192            }
193        }
194    }
195
196    let app_name = app.package_info().name.trim();
197    if app_name.is_empty() {
198        "Lumen Desktop".to_string()
199    } else {
200        app_name.to_string()
201    }
202}

388pub fn register_device(
389    app: &AppHandle,
390    state: &State<'_, DeviceState>,
391    payload: RegisterPayload,
392    sender: UnboundedSender<Message>,
393) -> Result<AuthOkResponse, String> {
394    let now = now_ts();
395    {
396        let mut pending = state.pending_token.lock().map_err(|e| e.to_string())?;
397        let token = pending
398            .as_mut()
399            .ok_or_else(|| "token_expired".to_string())?;
400
401        if token.expires_at < now {
402            return Err("token_expired".to_string());
403        }
404
405        if token.is_used {
406            return Err("token_used".to_string());
407        }
408
409        if token.token != payload.token {
410            return Err("token_expired".to_string());
411        }
412
413        token.is_used = true;
414    }
415
416    let existing = {
417        let devices = state.devices.lock().map_err(|e| e.to_string())?;
418        devices.get(&payload.device_id).cloned()
419    };
420
421    let device = Device {
422        device_id: payload.device_id,
423        device_name: payload.device_name,
424        device_type: payload.device_type,
425        os: payload.os,
426        version: payload.version,
427        access_token: Uuid::new_v4().to_string(),
428        is_active: true,
429        permissions: existing
430            .as_ref()
431            .map(|d| d.permissions.clone())
432            .unwrap_or_default(),
433        registered_at: existing.as_ref().map(|d| d.registered_at).unwrap_or(now),
434        last_connected_at: Some(now),
435    };
436
437    insert_or_replace_device_in_db(&device)?;
438
439    {
440        let mut devices = state.devices.lock().map_err(|e| e.to_string())?;
441        devices.insert(device.device_id.clone(), device.clone());
442    }
443
444    let session_id = create_or_replace_session(
445        state,
446        device.device_id.clone(),
447        device.permissions.clone(),
448        sender,
449    )?;
450
451    let event_name = if existing.is_some() {
452        "device_updated"
453    } else {
454        "device_registered"
455    };
456    let _ = app.emit(event_name, device.clone());
457    let _ = app.emit("device_authenticated", device.clone());
458
459    Ok(AuthOkResponse {
460        event: "auth_ok",
461        session_id,
462        desktop_name: resolve_desktop_name(app),
463        permissions: device.permissions.clone(),
464        access_token: Some(device.access_token),
465    })
466}

468pub fn authenticate_device(
469    app: &AppHandle,
470    state: &State<'_, DeviceState>,
471    payload: AuthPayload,
472    sender: UnboundedSender<Message>,
473) -> Result<AuthOkResponse, String> {
474    let settings = state
475        .remote_settings
476        .lock()
477        .map_err(|e| e.to_string())?
478        .clone();
479    if !settings.remote_enabled {
480        return Err("unauthorized".to_string());
481    }
482
483    let device = {
484        let devices = state.devices.lock().map_err(|e| e.to_string())?;
485        devices
486            .get(&payload.device_id)
487            .cloned()
488            .ok_or_else(|| "not_registered".to_string())?
489    };
490
491    if device.access_token != payload.access_token {
492        return Err("invalid_token".to_string());
493    }
494
495    if !device.is_active {
496        return Err("not_active".to_string());
497    }
498
499    let mut updated_device = device.clone();
500    updated_device.last_connected_at = Some(now_ts());
501    update_device_in_db(&updated_device)?;
502
503    {
504        let mut devices = state.devices.lock().map_err(|e| e.to_string())?;
505        devices.insert(updated_device.device_id.clone(), updated_device.clone());
506    }
507
508    let session_id = create_or_replace_session(
509        state,
510        updated_device.device_id.clone(),
511        updated_device.permissions.clone(),
512        sender,
513    )?;
514
515    let _ = app.emit("device_authenticated", updated_device.clone());
516
517    Ok(AuthOkResponse {
518        event: "auth_ok",
519        session_id,
520        desktop_name: resolve_desktop_name(app),
521        permissions: updated_device.permissions,
522        access_token: None,
523    })
524}

589pub fn broadcast_remote_event_inner(
590    state: &State<'_, DeviceState>,
591    envelope: &RemoteSyncEnvelope,
592    required_permission: Option<&str>,
593) -> Result<(), String> {
594    let message = json_message(envelope)?;
595    let sessions = state.sessions.lock().map_err(|e| e.to_string())?;
596
597    for session in sessions.values() {
598        if let Some(permission) = required_permission {
599            if !is_permission_allowed(&session.permissions, permission) {
600                continue;
601            }
602        }
603
604        if let Some(sender) = &session.sender {
605            let _ = sender.send(message.clone());
606        }
607    }
608
609    Ok(())
610}

648fn close_all_external_sessions(
649    app: &AppHandle,
650    state: &State<'_, DeviceState>,
651    auth_fail_reason: Option<String>,
652    close_code: Option<u16>,
653) -> Result<(), String> {
654    let device_ids = {
655        let sessions = state.sessions.lock().map_err(|e| e.to_string())?;
656        sessions
657            .values()
658            .map(|session| session.device_id.clone())
659            .collect::<Vec<_>>()
660    };
661
662    for device_id in device_ids {
663        close_device_sessions(
664            state,
665            &device_id,
666            auth_fail_reason
667                .as_deref()
668                .map(auth_fail_message)
669                .transpose()?,
670            close_code,
671        )?;
672    }
673
674    let _ = app.emit("remote_access_disabled", ());
675    Ok(())
676}

678fn close_device_sessions(
679    state: &State<'_, DeviceState>,
680    device_id: &str,
681    auth_fail_message: Option<Message>,
682    close_code: Option<u16>,
683) -> Result<(), String> {
684    let session_ids = {
685        let sessions = state.sessions.lock().map_err(|e| e.to_string())?;
686        sessions
687            .iter()
688            .filter_map(|(session_id, session)| {
689                if session.device_id == device_id {
690                    Some(session_id.clone())
691                } else {
692                    None
693                }
694            })
695            .collect::<Vec<_>>()
696    };
697
698    let mut sessions = state.sessions.lock().map_err(|e| e.to_string())?;
699    for session_id in session_ids {
700        if let Some(session) = sessions.remove(&session_id) {
701            if let Some(sender) = session.sender {
702                if let Some(message) = auth_fail_message.clone() {
703                    let _ = sender.send(message);
704                }
705                if let Some(code) = close_code {
706                    let _ = sender.send(close_message(code));
707                }
708            }
709        }
710    }
711
712    Ok(())
713}

751fn ensure_devices_schema(connection: &Connection) -> Result<(), String> {
752    let mut statement = connection
753        .prepare("PRAGMA table_info(devices)")
754        .map_err(|e| e.to_string())?;
755    let columns: Vec<String> = statement
756        .query_map([], |row| row.get::<_, String>(1))
757        .map_err(|e| e.to_string())?
758        .collect::<Result<Vec<_>, _>>()
759        .map_err(|e| e.to_string())?;
760
761    let has_permissions_streaming = columns.iter().any(|c| c == "permissions_streaming");
762
763    if !has_permissions_streaming {
764        connection
765            .execute(
766                "ALTER TABLE devices ADD COLUMN permissions_streaming INTEGER NOT NULL DEFAULT 0",
767                [],
768            )
769            .map_err(|e| e.to_string())?;
770    }
771
772    let has_permissions_chat = columns.iter().any(|c| c == "permissions_chat");
773
774    if !has_permissions_chat {
775        connection
776            .execute(
777                "ALTER TABLE devices ADD COLUMN permissions_chat INTEGER NOT NULL DEFAULT 0",
778                [],
779            )
780            .map_err(|e| e.to_string())?;
781    }
782
783    Ok(())
784}

821fn load_devices_from_db() -> Result<HashMap<String, Device>, String> {
822    let connection = open_device_db()?;
823    let mut statement = connection
824        .prepare(
825            r#"
826            SELECT
827                device_id,
828                device_name,
829                device_type,
830                os,
831                version,
832                access_token,
833                is_active,
834                permissions_player,
835                permissions_lyrics,
836                permissions_bible,
837                permissions_media,
838                permissions_streaming,
839                permissions_chat,
840                registered_at,
841                last_connected_at
842            FROM devices
843            "#,
844        )
845        .map_err(|e| e.to_string())?;
846
847    let rows = statement
848        .query_map([], |row| {
849            Ok(Device {
850                device_id: row.get(0)?,
851                device_name: row.get(1)?,
852                device_type: row.get(2)?,
853                os: row.get(3)?,
854                version: row.get(4)?,
855                access_token: row.get(5)?,
856                is_active: row.get::<_, i64>(6)? == 1,
857                permissions: DevicePermissions {
858                    player: row.get::<_, i64>(7)? == 1,
859                    lyrics: row.get::<_, i64>(8)? == 1,
860                    bible: row.get::<_, i64>(9)? == 1,
861                    media: row.get::<_, i64>(10)? == 1,
862                    streaming: row.get::<_, i64>(11)? == 1,
863                    chat: row.get::<_, i64>(12)? == 1,
864                },
865                registered_at: row.get::<_, u64>(13)?,
866                last_connected_at: row.get(14)?,
867            })
868        })
869        .map_err(|e| e.to_string())?;
870
871    let mut devices = HashMap::new();
872    for row in rows {
873        let device = row.map_err(|e| e.to_string())?;
874        devices.insert(device.device_id.clone(), device);
875    }
876
877    Ok(devices)
878}

950pub fn map_event_permission(event: &str) -> Option<&'static str> {
951    match event {
952        "play_pause" | "stop" | "next" | "previous" | "mute" | "set_volume" | "seek"
953        | "set_loop" | "load_url" | "metadata" | "progress" => Some("player"),
954        "load_lyric" => Some("lyrics"),
955        "subscribe_stream"
956        | "unsubscribe_stream"
957        | "webrtc_answer"
958        | "webrtc_ice_candidate"
959        | "mobile_offer" => Some("streaming"),
960        "chat_send" | "chat_file_send" | "chat_history" | "chat_reaction" | "chat_typing" | "chat_read" => Some("chat"),
961        _ => None,
962    }
963}
