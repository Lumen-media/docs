1//! Media download subsystem: external tool provisioning, video downloads, and
2//! YouTube cookie management.
3//!
4//! Three concerns live behind this module, each in its own submodule:
5//!
6//! - [`dependencies`] - detects, downloads, and versions the external tools
7//!   (`yt-dlp`, `ffmpeg`/`ffprobe`, `node`) under `<app-data>/tools`.
8//! - [`downloader`] - spawns `yt-dlp` to download videos/audio, streams
9//!   progress events, and manages cancellation and cookie validation.
10//! - [`github`] - resolves release asset URLs from GitHub/Node.js endpoints.
11//!
12//! [`DownloadState`] is created once during app setup with
13//! [`initialize_download_state`] and registered with Tauri's state manager so
14//! every command below can reach the shared tools directory and the set of
15//! active downloads.
16//!
17//! # Frontend commands
18//!
19//! | Command | Invoke name | Returns / Events |
20//! |---|---|---|
21//! | [`check_dependencies`] | `check_dependencies` | [`DependencyStatus`] |
22//! | [`download_dependencies`] | `download_dependencies` | emits `dependency-download-progress` / `dependency-download-complete` |
23//! | [`list_dependencies`] | `list_dependencies` | `DependencyInfo[]` |
24//! | [`download_video`] | `download_video` | initial [`DownloadResult`] plus `video-download-progress` / `video-download-complete` / `video-download-error` events (see [`downloader::start_download`]) |
25//! | [`cancel_download`] | `cancel_download` | kills the underlying `yt-dlp` process |
26//! | [`get_download_status`] | `get_download_status` | [`DownloadStatusInfo`] or `null` |
27//! | [`install_cookies_file`] | `install_cookies_file` | installed path as `string` |
28//! | [`validate_cookies`] | `validate_cookies` | [`downloader::CookieValidation`] |
29

31pub mod downloader;

42/// Shared, Tauri-managed state for the download subsystem.
43///
44/// Holds the directory where downloaded tools live and the map of in-flight
45/// downloads that commands mutate through a locked handle ([`downloader::ActiveDownloads`]).
46#[derive(Clone)]
47pub struct DownloadState {
48    /// Directory containing the provisioned tools (`yt-dlp`, `ffmpeg`,
49    /// `node/`, `cookies.txt`). Usually `<app-data>/tools`.
50    pub tools_dir: PathBuf,
51    /// In-flight downloads keyed by download id, shared with the background
52    /// progress tasks spawned by [`download_video`].
53    pub active_downloads: downloader::ActiveDownloads,
54}
55
56/// Lightweight snapshot of a download's state returned by
57/// [`get_download_status`].
58#[derive(Debug, Clone, Serialize)]
59pub struct DownloadStatusInfo {
60    /// Current status string; currently always `"downloading"` when the
61    /// download is present in the active map.
62    pub status: String,
63    /// Completion fraction in `0.0..=1.0`, or `None` when unknown. Progress is
64    /// pushed through the `video-download-progress` event instead.
65    pub progress: Option<f64>,
66}

68/// Resolves the tools directory from the app data path.
69fn tools_dir_from_app(app: &AppHandle) -> Result<PathBuf, String> {
70    let data_dir = app
71        .path()
72        .app_data_dir()
73        .map_err(|e| format!("Failed to get app data dir: {}", e))?;
74    Ok(data_dir.join("tools"))
75}

77/// Creates the tools directory and the [`DownloadState`] it depends on.
78///
79/// Called once during app setup; the returned value is registered with
80/// `app.manage(...)` so the Tauri commands can request it via
81/// `State<DownloadState>`.
82///
83/// # Errors
84///
85/// Returns an error string when the app data directory cannot be resolved or
86/// the tools directory cannot be created.
87pub fn initialize_download_state(app: &AppHandle) -> Result<DownloadState, String> {
88    let dir = tools_dir_from_app(app)?;
89    std::fs::create_dir_all(&dir)
90        .map_err(|e| format!("Failed to create tools directory: {}", e))?;
91
92    Ok(DownloadState {
93        tools_dir: dir,
94        active_downloads: downloader::create_active_downloads(),
95    })
96}

98/// Reports which external dependencies are installed and whether `yt-dlp` is
99/// outdated.
100///
101/// # Tauri command (frontend API)
102///
103/// `invoke('check_dependencies')` resolves with a [`DependencyStatus`]
104/// describing `yt-dlp`, `ffmpeg`, cookies, and `node`, plus the resolved tools
105/// directory. When `yt-dlp` is installed the latest GitHub release is fetched
106/// to compute `ytdlp_outdated`; network failures there are ignored and simply
107/// leave the flag `false`.
108///
109/// # Errors
110///
111/// Returns an error string when the app data directory cannot be resolved.
112#[tauri::command]
113pub async fn check_dependencies(
114    _state: State<'_, DownloadState>,
115    app: AppHandle,
116) -> Result<DependencyStatus, String> {
117    dependencies::check_dependencies(&app).await
118}

120/// Downloads or updates the external tools, emitting progress events.
121///
122/// # Tauri command (frontend API)
123///
124/// `invoke('download_dependencies')` resolves once provisioning finishes (or
125/// fails). Progress is reported through the `dependency-download-progress`
126/// event (`{ tool, progress, status, bytes_downloaded?, total_bytes? }`) and
127/// completion through `dependency-download-complete` (`{ tool, version }`).
128///
129/// `ffmpeg` and `node` are only downloaded when missing; `yt-dlp` is updated
130/// whenever the installed version differs from the latest release. On macOS
131/// the FFmpeg download is skipped and FFmpeg must be installed manually.
132///
133/// # Errors
134///
135/// Returns an error string when the app data directory cannot be resolved, a
136/// release cannot be fetched, a download fails, or an archive cannot be
137/// extracted.
138#[tauri::command]
139pub async fn download_dependencies(app: AppHandle) -> Result<(), String> {
140    dependencies::download_dependencies(app).await
141}

159/// Starts a `yt-dlp` download for `url` and returns immediately.
160///
161/// # Tauri command (frontend API)
162///
163/// `invoke('download_video', { url, provider, quality, maxHeight? })` returns
164/// a [`DownloadResult`] whose `download_id` identifies the job; the other
165/// fields are placeholders (`file_path`/`file_extension` empty, `file_size`
166/// `0`) until the download completes. The real outcome arrives through events:
167///
168/// - `video-download-progress` - `{ download_id, progress, speed, eta, status }`
169/// - `video-download-complete` - the finished [`DownloadResult`]
170/// - `video-download-error` - `{ download_id, message, code }`
171///
172/// # Errors
173///
174/// Returns an error string when the provider is not `youtube`, `yt-dlp` is not
175/// installed, FFmpeg is missing on macOS, or the downloads directory cannot be
176/// created or the process cannot be spawned. Behaviour is otherwise delegated
177/// to [`downloader::start_download`].
178#[tauri::command]
179pub async fn download_video(
180    state: State<'_, DownloadState>,
181    app: AppHandle,
182    url: String,
183    provider: String,
184    quality: DownloadQuality,
185    max_height: Option<u32>,
186) -> Result<DownloadResult, String> {
187    downloader::start_download(
188        app,
189        state.tools_dir.clone(),
190        state.active_downloads.clone(),
191        url,
192        provider,
193        quality,
194        max_height,
195    )
196    .await
197}

199/// Cancels an in-flight download by killing its `yt-dlp` process.
200///
201/// # Tauri command (frontend API)
202///
203/// `invoke('cancel_download', { downloadId })` removes the download from the
204/// active map and terminates the process. Resolves even when the id is
205/// unknown; no completion/error event is emitted for a cancelled job.
206///
207/// # Errors
208///
209/// Returns an error string only when the process exists but cannot be killed.
210#[tauri::command]
211pub async fn cancel_download(
212    state: State<'_, DownloadState>,
213    download_id: String,
214) -> Result<(), String> {
215    downloader::cancel_download(&state.active_downloads, &download_id).await
216}

218/// Returns whether a download id is currently active.
219///
220/// # Tauri command (frontend API)
221///
222/// `invoke('get_download_status', { downloadId })` resolves with
223/// [`DownloadStatusInfo`] while the job is running or `null` once it has
224/// finished or been cancelled. Detailed progress is only available through the
225/// `video-download-progress` event.
226///
227/// # Errors
228///
229/// Currently never fails, but returns a `Result` to match the Tauri command
230/// boundary.
231#[tauri::command]
232pub async fn get_download_status(
233    state: State<'_, DownloadState>,
234    download_id: String,
235) -> Result<Option<DownloadStatusInfo>, String> {
236    let downloads = state.active_downloads.lock().await;
237    if downloads.contains_key(&download_id) {
238        Ok(Some(DownloadStatusInfo {
239            status: "downloading".to_string(),
240            progress: None,
241        }))
242    } else {
243        Ok(None)
244    }
245}

247/// Installs a user-provided Netscape-format `cookies.txt` into the tools
248/// directory.
249///
250/// # Tauri command (frontend API)
251///
252/// `invoke('install_cookies_file', { sourcePath })` validates the file is a
253/// Netscape cookie jar (correct header and at least one tab-separated cookie
254/// line) and copies it to `<tools-dir>/cookies.txt`, which `yt-dlp` then uses
255/// for YouTube authentication. Resolves with the destination path.
256///
257/// The file is expected to come from the "Get cookies.txt LOCALLY" browser
258/// extension with the user logged in to YouTube.
259///
260/// # Errors
261///
262/// Returns an error string when the source path does not exist, the file
263/// cannot be read or written, or the content does not look like a valid
264/// Netscape cookie file.
265#[tauri::command]
266pub async fn install_cookies_file(
267    state: State<'_, DownloadState>,
268    source_path: String,
269) -> Result<String, String> {
270    let source = std::path::Path::new(&source_path);
271    if !source.exists() {
272        return Err("Arquivo de cookies não encontrado".to_string());
273    }
274
275    let content = std::fs::read(source)
276        .map_err(|e| format!("Falha ao ler o arquivo de cookies: {}", e))?;
277
278    // Netscape cookie files always start with this header line.
279    let header = b"# Netscape HTTP Cookie File";
280    if !content.starts_with(header) {
281        return Err(
282            "O arquivo não parece ser um cookies.txt do YouTube (formato Netscape inválido). Verifique se você exportou com a extensão \"Get cookies.txt LOCALLY\".".to_string()
283        );
284    }
285
286    // Ensure at least one non-comment, tab-separated cookie line exists.
287    let text = String::from_utf8_lossy(&content);
288    let has_cookie_line = text.lines().any(|line| {
289        let line = line.trim();
290        !line.is_empty() && !line.starts_with('#') && line.contains('\t')
291    });
292    if !has_cookie_line {
293        return Err(
294            "O arquivo de cookies está vazio ou não contém cookies válidos. Faça login no YouTube e exporte novamente.".to_string()
295        );
296    }
297
298    let dest = state.tools_dir.join("cookies.txt");
299    std::fs::write(&dest, &content)
300        .map_err(|e| format!("Falha ao salvar cookies: {}", e))?;
301
302    Ok(dest.to_string_lossy().to_string())
303}

305/// Validates the installed cookies (or a detected logged-in browser) against
306/// YouTube.
307///
308/// # Tauri command (frontend API)
309///
310/// `invoke('validate_cookies')` resolves with a
311/// [`downloader::CookieValidation`] whose `status` is one of `valid`,
312/// `missing`, `error`, `rotated`, `blocked`, or `no_account`, with a
313/// human-readable `detail`. When no `cookies.txt` exists this also tries to
314/// detect cookies from a logged-in browser.
315///
316/// # Errors
317///
318/// Returns an error string when `yt-dlp` is not installed or the validation
319/// process cannot be spawned.
320#[tauri::command]
321pub async fn validate_cookies(
322    state: State<'_, DownloadState>,
323) -> Result<downloader::CookieValidation, String> {
324    downloader::validate_cookies(&state.tools_dir).await
