1//! Recursive listing of media folders for the frontend.
2//!
3//! The file browser receives a flat, alphabetically sorted list of every file
4//! under a media type's folder, carrying enough metadata (size, modification
5//! time, extension, subfolder) to filter, group and re-sort on the client
6//! without further disk access.
7
8use serde::Serialize;
9use std::path::Path;
10
11/// A single file discovered by [`scan_media_files`].
12#[derive(Serialize)]
13#[serde(rename_all = "camelCase")]
14pub struct ScannedFile {
15    /// File name including its extension.
16    pub name: String,
17    /// Absolute path of the file.
18    pub path: String,
19    /// Folder containing the file, relative to the media root; empty for
20    /// files sitting at the root itself.
21    pub folder: String,
22    /// File size in bytes.
23    pub size: i64,
24    /// Last modification time in Unix milliseconds, `0` when unavailable.
25    pub modified_at: i64,
26    /// Extension without the leading dot, or an empty string.
27    pub extension: String,
28}
29
30/// Resolves the root folder of a media type through [`crate::paths`].
31fn media_type_dir(media_type: &str) -> Result<std::path::PathBuf, String> {
32    crate::paths::media_dir(media_type)
33}
34
35/// Extension of `name` without the leading dot, or an empty string when the
36/// name carries none.
37fn ext_from_name(name: &str) -> String {
38    Path::new(name)
39        .extension()
40        .map(|e| e.to_string_lossy().to_string())
41        .unwrap_or_default()
42}
43
44/// Lists every file under a media type's folder for the frontend.
45///
46/// The tree is walked depth-first; symlinks and other non-regular entries are
47/// skipped, and results are sorted case-insensitively by name. A media folder
48/// that does not exist resolves to an empty list instead of an error.
49///
50/// # Errors
51///
52/// Returns an error when the media type is unknown or paths have not been
53/// initialized, or when a directory, an entry type or file metadata cannot be
54/// read.
55#[tauri::command]
56pub async fn scan_media_files(media_type: String) -> Result<Vec<ScannedFile>, String> {
57    let dir = media_type_dir(&media_type)?;
58    if !dir.is_dir() {
59        return Ok(Vec::new());
60    }
61
62    let mut files: Vec<ScannedFile> = Vec::new();
63    let mut stack: Vec<(std::path::PathBuf, String)> = vec![(dir.clone(), String::new())];
64
65    while let Some((current_dir, folder)) = stack.pop() {
66        let entries = std::fs::read_dir(&current_dir).map_err(|e| e.to_string())?;
67
68        for entry in entries.flatten() {
69            let file_type = entry.file_type().map_err(|e| e.to_string())?;
70            let path = entry.path();
71            let name = entry.file_name().to_string_lossy().to_string();
72
73            if file_type.is_dir() {
74                let child = if folder.is_empty() {
75                    name.clone()
76                } else {
77                    format!("{folder}/{name}")
78                };
79                stack.push((path, child));
80                continue;
81            }
82            if !file_type.is_file() {
83                continue;
84            }
85
86            let meta = std::fs::metadata(&path).map_err(|e| e.to_string())?;
87
88            files.push(ScannedFile {
89                name: name.clone(),
90                path: path.to_string_lossy().to_string(),
91                folder: folder.clone(),
92                size: meta.len() as i64,
93                modified_at: meta
94                    .modified()
95                    .ok()
96                    .and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok())
97                    .map(|d| d.as_millis() as i64)
98                    .unwrap_or(0),
99                extension: ext_from_name(&name),
100            });
101        }
102    }
103
104    files.sort_by(|a, b| a.name.to_lowercase().cmp(&b.name.to_lowercase()));
105    Ok(files)
106}
