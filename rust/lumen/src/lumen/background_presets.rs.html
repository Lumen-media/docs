1//! Generator presets owned by Rust.
2//!
3//! One pretty-printed `<id>.json` file per user preset lives under
4//! `<base>/config/background-presets/`; built-ins are defined in code and
5//! merged with the user list. Every parameter is validated and clamped
6//! against the catalog before it is persisted, so the webview can never
7//! introduce an effect or a value the engine does not understand.
8

18/// A saved generator configuration.
19#[derive(Debug, Clone, Serialize, Deserialize)]
20#[serde(rename_all = "camelCase")]
21pub struct GeneratorPreset {
22    /// Stable identifier; also the file name for user presets.
23    pub id: String,
24    /// Display name chosen by the user.
25    pub name: String,
26    /// Effect id from the Rust catalog.
27    pub effect: String,
28    /// Parameter overrides keyed by catalog key.
29    #[serde(default)]
30    pub params: BTreeMap<String, Value>,
31    /// Reproducible variation seed.
32    #[serde(default)]
33    pub seed: u32,
34    /// Creation time in Unix milliseconds, used to order listings.
35    #[serde(default)]
36    pub created_at: i64,
37    /// Catalog version the preset was authored against.
38    #[serde(default = "default_version")]
39    pub version: u32,
40    /// Built-ins ship with the app and cannot be edited or deleted.
41    #[serde(default)]
42    pub built_in: bool,
43}
44
45fn default_version() -> u32 {
46    1
47}

49fn presets_dir() -> Result<PathBuf, String> {
50    let dir = crate::remote::app_base_dir()?
51        .join("config")
52        .join("background-presets");
53    std::fs::create_dir_all(&dir).map_err(|error| error.to_string())?;
54    Ok(dir)
55}

57fn params(entries: &[(&str, f64)]) -> BTreeMap<String, Value> {
58    entries
59        .iter()
60        .map(|(key, value)| ((*key).to_string(), serde_json::json!(value)))
61        .collect()
62}

120fn is_hex_color(value: &str) -> bool {
121    value.len() == 7
122        && value.starts_with('#')
123        && value[1..].chars().all(|c| c.is_ascii_hexdigit())
124}
125
126fn safe_id(id: &str) -> Result<&str, String> {
127    if id.is_empty()
128        || !id
129            .chars()
130            .all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '_')
131    {
132        return Err("invalid preset id".to_string());
133    }
134    Ok(id)
135}

137/// Fills defaults, drops unknown keys and clamps values to the catalog.
138fn normalize(
139    effect_id: &str,
140    values: &BTreeMap<String, Value>,
141    seed: u32,
142) -> Result<(BTreeMap<String, Value>, u32), String> {
143    let effect = catalog::catalog()
144        .into_iter()
145        .find(|effect| effect.id == effect_id)
146        .ok_or_else(|| format!("unknown generator effect `{effect_id}`"))?;
147
148    let seed = seed.min(9999);
149    let mut normalized = BTreeMap::new();
150
151    for spec in &effect.params {
152        let raw = values.get(spec.key);
153        let value = match spec.kind {
154            ParamKind::Color => {
155                let fallback = match &spec.default {
156                    ParamValue::Text(text) => text.clone(),
157                    _ => "#000000".to_string(),
158                };
159                let color = raw
160                    .and_then(Value::as_str)
161                    .filter(|value| is_hex_color(value))
162                    .map(str::to_string)
163                    .unwrap_or(fallback);
164                Value::String(color)
165            }
166            ParamKind::Bool => {
167                let fallback = matches!(spec.default, ParamValue::Bool(true));
168                Value::Bool(raw.and_then(Value::as_bool).unwrap_or(fallback))
169            }
170            ParamKind::Seed => serde_json::json!(seed),
171            _ => {
172                let fallback = match spec.default {
173                    ParamValue::Number(number) => number,
174                    _ => 0.0,
175                };
176                let mut number = raw.and_then(Value::as_f64).unwrap_or(fallback);
177                if let Some(min) = spec.min {
178                    number = number.max(min);
179                }
180                if let Some(max) = spec.max {
181                    number = number.min(max);
182                }
183                if matches!(spec.kind, ParamKind::Int) {
184                    number = number.round();
185                }
186                serde_json::json!(number)
187            }
188        };
189        normalized.insert(spec.key.to_string(), value);
190    }
191
192    Ok((normalized, seed))
193}

195/// Returns built-ins plus every valid user preset, newest first.
196#[tauri::command]
197pub fn bg_preset_list() -> Result<Vec<GeneratorPreset>, String> {
198    let mut presets = builtin_presets();
199
200    for entry in std::fs::read_dir(presets_dir()?)
201        .map_err(|error| error.to_string())?
202        .flatten()
203    {
204        if !entry.file_name().to_string_lossy().ends_with(".json") {
205            continue;
206        }
207        let Ok(raw) = std::fs::read_to_string(entry.path()) else {
208            continue;
209        };
210        let Ok(preset) = serde_json::from_str::<GeneratorPreset>(&raw) else {
211            continue;
212        };
213        // A hand-written file must not take over a built-in's id.
214        if preset.id.starts_with("builtin-") {
215            continue;
216        }
217        presets.retain(|candidate| candidate.id != preset.id);
218        presets.push(preset);
219    }
220
221    presets.sort_by(|a, b| b.created_at.cmp(&a.created_at));
222    Ok(presets)
223}

225/// Creates or overwrites a user preset after validating it against the catalog.
226#[tauri::command]
227pub fn bg_preset_save(mut preset: GeneratorPreset) -> Result<GeneratorPreset, String> {
228    if preset.built_in {
229        return Err("built-in presets cannot be overwritten".to_string());
230    }
231    if preset.name.trim().is_empty() {
232        return Err("preset name is required".to_string());
233    }
234    if preset.id.trim().is_empty() {
235        preset.id = uuid::Uuid::new_v4().to_string();
236    }
237    safe_id(&preset.id)?;
238    if preset.id.starts_with("builtin-") {
239        return Err("built-in presets cannot be overwritten".to_string());
240    }
241
242    let (params, seed) = normalize(&preset.effect, &preset.params, preset.seed)?;
243    preset.params = params;
244    preset.seed = seed;
245    preset.version = 1;
246    if preset.created_at == 0 {
247        preset.created_at = crate::remote::now_ms();
248    }
249
250    let path = presets_dir()?.join(format!("{}.json", preset.id));
251    let raw = serde_json::to_string_pretty(&preset).map_err(|error| error.to_string())?;
252    std::fs::write(path, raw).map_err(|error| error.to_string())?;
253    Ok(preset)
254}

256/// Deletes a user preset. Missing files and built-ins are left untouched.
257#[tauri::command]
258pub fn bg_preset_delete(id: String) -> Result<(), String> {
259    if id.starts_with("builtin-") {
260        return Ok(());
261    }
262    safe_id(&id)?;
263    let path = presets_dir()?.join(format!("{id}.json"));
264    if path.exists() {
265        std::fs::remove_file(path).map_err(|error| error.to_string())?;
266    }
267    Ok(())
268}
