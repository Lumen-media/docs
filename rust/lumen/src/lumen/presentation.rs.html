1//! Metadata extraction for PowerPoint presentations.
2//!
3//! A `.pptx` file is a ZIP archive of XML parts. The module walks
4//! `ppt/slides/slide<N>.xml` in numeric order, collects the text runs of
5//! each slide and reads the presentation title, so the frontend can index
6//! and search presentations without opening them.
7
8use quick_xml::events::Event;
9use quick_xml::Reader;
10use serde::Serialize;
11use std::io::Read;
12use zip::ZipArchive;
13
14/// Text content of a single slide.
15#[derive(Serialize)]
16pub struct SlideText {
17    /// Slide number as encoded in the archive entry name (`slide1.xml` is 1).
18    pub index: u32,
19    /// All text runs of the slide, joined with spaces.
20    pub text: String,
21}
22
23/// Summary of a presentation returned to the frontend.
24#[derive(Serialize)]
25pub struct PresentationMeta {
26    /// Number of slide entries found in the archive.
27    pub slide_count: u32,
28    /// Per-slide text, ordered by slide number.
29    pub slides: Vec<SlideText>,
30    /// Title from `ppt/presentation.xml`, falling back to the file name
31    /// without extension.
32    pub title: Option<String>,
33}
34
35/// Collects every `a:t` text run of a slide's XML, joined with spaces.
36///
37/// A malformed document ends the scan early and yields the text gathered so
38/// far instead of failing the extraction.
39fn extract_text_from_xml(xml: &[u8]) -> String {
40    let mut reader = Reader::from_reader(xml);
41    reader.config_mut().trim_text(true);
42    let mut buf = Vec::new();
43    let mut text = String::new();
44    let mut in_text_tag = false;
45
46    loop {
47        match reader.read_event_into(&mut buf) {
48            Ok(Event::Start(ref e)) | Ok(Event::Empty(ref e)) => {
49                if e.name().as_ref() == b"a:t" {
50                    in_text_tag = true;
51                }
52            }
53            Ok(Event::Text(ref e)) if in_text_tag => {
54                if let Ok(t) = e.unescape() {
55                    text.push_str(&t);
56                    text.push(' ');
57                }
58            }
59            Ok(Event::End(ref e)) => {
60                if e.name().as_ref() == b"a:t" {
61                    in_text_tag = false;
62                }
63            }
64            Ok(Event::Eof) => break,
65            Err(e) => {
66                eprintln!("XML parse error: {e}");
67                break;
68            }
69            _ => {}
70        }
71        buf.clear();
72    }
73
74    text.trim().to_string()
75}

77/// Reads the first non-empty text node inside the `presentation` root.
78fn extract_title(xml: &[u8]) -> Option<String> {
79    let mut reader = Reader::from_reader(xml);
80    reader.config_mut().trim_text(true);
81    let mut buf = Vec::new();
82    let mut depth = 0;
83
84    loop {
85        match reader.read_event_into(&mut buf) {
86            Ok(Event::Start(ref e)) => {
87                let name = e.name();
88                if name.as_ref() == b"p:presentation" || name.as_ref() == b"presentation" {
89                    depth += 1;
90                }
91            }
92            Ok(Event::End(ref e)) => {
93                let name = e.name();
94                if name.as_ref() == b"p:presentation" || name.as_ref() == b"presentation" {
95                    depth -= 1;
96                }
97            }
98            Ok(Event::Text(ref e)) if depth > 0 => {
99                if let Ok(t) = e.unescape() {
100                    let trimmed = t.trim();
101                    if !trimmed.is_empty() {
102                        return Some(trimmed.to_string());
103                    }
104                }
105            }
106            Ok(Event::Eof) => break,
107            Err(_) => break,
108            _ => {}
109        }
110        buf.clear();
111    }
112
113    None
114}

116/// Extracts slide text and title from a `.pptx` file for the frontend.
117///
118/// Slides are discovered by entry name, sorted numerically and read one by
119/// one; a slide that cannot be read is reported on stderr and skipped. The
120/// title falls back to the file name without extension when the archive does
121/// not carry a usable one.
122///
123/// # Errors
124///
125/// Returns an error when the file cannot be opened, is not a readable ZIP
126/// archive, or a slide's XML cannot be read out of it.
127#[tauri::command]
128pub fn extract_presentation_metadata(path: String) -> Result<PresentationMeta, String> {
129    let file = std::fs::File::open(&path)
130        .map_err(|e| format!("Failed to open file: {e}"))?;
131
132    let mut archive = ZipArchive::new(file)
133        .map_err(|e| format!("Failed to read ZIP archive: {e}"))?;
134
135    let mut slides: Vec<SlideText> = Vec::new();
136
137    // Sort slide entries by index
138    let mut slide_indices: Vec<u32> = Vec::new();
139    for i in 0..archive.len() {
140        let entry = archive.by_index(i).map_err(|e| format!("ZIP read error: {e}"))?;
141        let name = entry.name().to_string();
142        if let Some(idx) = parse_slide_index(&name) {
143            slide_indices.push(idx);
144        }
145    }
146    slide_indices.sort();
147
148    for idx in &slide_indices {
149        let entry_name = format!("ppt/slides/slide{}.xml", idx);
150        match archive.by_name(&entry_name) {
151            Ok(mut entry) => {
152                let mut xml = Vec::new();
153                entry.read_to_end(&mut xml).map_err(|e| format!("Failed to read slide XML: {e}"))?;
154                let text = extract_text_from_xml(&xml);
155                slides.push(SlideText {
156                    index: *idx,
157                    text,
158                });
159            }
160            Err(e) => {
161                eprintln!("Warning: could not read {entry_name}: {e}");
162            }
163        }
164    }
165
166    let slide_count = slide_indices.len() as u32;
167
168    let title = archive
169        .by_name("ppt/presentation.xml")
170        .ok()
171        .and_then(|mut entry| {
172            let mut xml = Vec::new();
173            entry.read_to_end(&mut xml).ok()?;
174            Some(extract_title(&xml))
175        })
176        .flatten()
177        // Fallback to file name without extension
178        .or_else(|| {
179            std::path::Path::new(&path)
180                .file_stem()
181                .and_then(|s| s.to_str())
182                .map(|s| s.to_string())
183        });
184
185    Ok(PresentationMeta {
186        slide_count,
187        slides,
188        title,
189    })
190}

192/// Extracts the number from an entry named `slide<N>.xml`, tolerating
193/// Windows path separators.
194fn parse_slide_index(name: &str) -> Option<u32> {
195    // Match patterns like: ppt/slides/slide1.xml or ppt/slides/slide01.xml
196    let name = name.replace('\\', "/");
197    let stem = std::path::Path::new(&name).file_stem()?.to_str()?;
198    if !stem.starts_with("slide") {
199        return None;
200    }
201    stem.trim_start_matches("slide").parse::<u32>().ok()
202}
